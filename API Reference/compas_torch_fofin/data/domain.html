    class Categories(DomainBase):
    """Represents a set of categorical categories."""

    def __init__(self, categories, dtype="categorical"):
        """
        Initialize the Categories instance with given categories and data type.

        Args:
            categories (list or array-like): The list of categories.
            dtype (str): The type of the categories, default is "categorical".
        """
        super().__init__()
        self.type = self.validate_type(dtype)
        self.categories = torch.tensor(categories)  # Options are now a PyTorch tensor

    def to_json(self):
        """
        Convert the Categories instance to a JSON-serializable dictionary.

        Returns:
            dict: A dictionary representation of the Categories instance.
        """
        return {"type": self.type, "categories": self.categories.tolist()}

    @classmethod
    def from_json(cls, data):
        """
        Create a Categories instance from a JSON-serializable dictionary.

        Args:
            data (dict): A dictionary containing the categories data.

        Returns:
            Categories: An instance of Categories with the data from the dictionary.
        """
        return cls(data['categories'], data['type'])

    def sample(self, size=1):
        """
        Generate random samples from the categories.

        Args:
            size (int): The number of samples to generate. Default is 1.

        Returns:
            torch.Tensor: A tensor containing the sampled categories.
        """
        indices = torch.randint(0, len(self.categories), (size,))
        return self.categories[indices]

    def copy(self):
        """
        Create a copy of the Categories instance.

        Returns:
            Categories: A new instance of Categories with the same data.
        """
        return self.__class__(self.categories.tolist(), self.type)

    def __repr__(self):
        """
        String representation of the Categories instance.

        Returns:
            str: A string representation of the Categories instance.
        """
        return f"{self.__class__.__name__}({self.categories.tolist()})"

    @staticmethod
    def validate_categories(categories):
        """
        Validate the categories to ensure they are not empty and are unique.

        Args:
            categories (list or array-like): The list of categories to validate.

        Raises:
            ValueError: If the categories list is empty or contains duplicates.
        """
        if not categories:
            raise ValueError("Options list cannot be empty.")
        # Ensure all categories are unique
        tensor_categories = torch.tensor(categories)
        _, indices = torch.unique(tensor_categories, sorted=True, return_inverse=True)
        unique_categories = tensor_categories[indices]
        return unique_categories.tolist()

    @staticmethod
    def validate_type(type_):
        """
        Validate the type to ensure it is one of the allowed types.

        Args:
            type_ (str): The type to validate.

        Returns:
            str: The validated type.
        """
        if type_ not in ["categorical", "ordinal", "integer", "any"]:
            raise ValueError(f"Invalid type '{type_}'. Must be 'categorical', 'ordinal', 'integer', or 'any'.")
        return type_

    class Interval(DomainBase):
    """Defines a numerical range."""

    def __init__(self, min_val=0, max_val=1, dtype="float"):
        """
        Initialize the Interval instance with a minimum value, maximum value, and data type.

        Args:
            min_val (int or float): The minimum value of the interval. Default is 0.
            max_val (int or float): The maximum value of the interval. Default is 1.
            dtype (str): The data type of the interval values. Default is "float".
        """
        super().__init__()
        self.type = self.validate_type(dtype)
        self.min_val, self.max_val = _cast_data(min_val, dtype=dtype), _cast_data(max_val, dtype=dtype)

    def to_json(self):
        """
        Convert the Interval instance to a JSON-serializable dictionary.

        Returns:
            dict: A dictionary representation of the Interval instance.
        """
        return {"type": self.type, "min_val": self.min_val, "max_val": self.max_val}

    @classmethod
    def from_json(cls, data):
        """
        Create an Interval instance from a JSON-serializable dictionary.

        Args:
            data (dict): A dictionary containing the interval data.

        Returns:
            Interval: An instance of Interval with the data from the dictionary.
        """
        return cls(data['min_val'], data['max_val'], data['type'])

    def sample(self, size=1):
        """
        Generate random samples from the interval.

        Args:
            size (int): The number of samples to generate. Default is 1.

        Returns:
            torch.Tensor: A tensor containing the sampled values.
        """
        if self.type == "int":
            return torch.randint(self.min_val, self.max_val + 1, (size,))
        elif self.type == "float":
            return torch.rand(size) * (self.max_val - self.min_val) + self.min_val

    def copy(self):
        """
        Create a copy of the Interval instance.

        Returns:
            Interval: A new instance of Interval with the same data.
        """
        return self.__class__(self.min_val, self.max_val, self.type)

    def __repr__(self):
        """
        String representation of the Interval instance.

        Returns:
            str: A string representation of the Interval instance.
        """
        return f"{self.__class__.__name__}({self.min_val}, {self.max_val})"

    @staticmethod
    def validate_range(min_val, max_val):
        """
        Validate the range to ensure the minimum value is less than the maximum value.

        Args:
            min_val (int or float): The minimum value of the range.
            max_val (int or float): The maximum value of the range.

        Returns:
            tuple: A tuple containing the validated minimum and maximum values.
        """
        if min_val > max_val:
            raise ValueError("Minimum value must be less than maximum value.")
        return min_val, max_val

    @staticmethod
    def validate_type(type_):
        """
        Validate the type to ensure it is one of the allowed types.

        Args:
            type_ (str): The type to validate.

        Returns:
            str: The validated type.
        """
        if type_ not in ["float", "int", "any"]:
            raise ValueError(f"Invalid type '{type_}'. Must be 'real', 'integer', or 'any'.")
        return type_

    def sample(self, size=1):
    """
    Generate random samples from the interval.

    Args:
        size (int): The number of samples to generate. Default is 1.

    Returns:
        torch.Tensor: A tensor containing the sampled values.
    """
    if self.type == "int":
        return torch.randint(self.min_val, self.max_val + 1, (size,))
    elif self.type == "float":
        return torch.rand(size) * (self.max_val - self.min_val) + self.min_val
