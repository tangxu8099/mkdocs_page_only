    def default_collate(batch):
    """
    Puts each data field into a tensor with outer dimension batch size.

    Args:
        batch (list): A list of data samples to be collated.

    Returns:
        The collated batch, which can be a tensor, list, dictionary, or other 
        data structure depending on the input data type.
    """
    elem = batch[0]
    elem_type = type(elem)

    if isinstance(elem, (th.Tensor, 
                         CollateByCat, CollateByStack)):

        out = None
        if not isinstance(elem, th.Tensor):
            assert len(set([type(_elem) for _elem in batch])) == 1
        _elem = elem

        elem = release_tensor(elem)
        batch = [release_tensor(_x) for _x in batch]

        if th.utils.data.get_worker_info() is not None:
            # If we're in a background process, concatenate directly into a
            # shared memory tensor to avoid an extra copy
            numel = sum([x.numel() for x in batch])
            storage = elem.storage()._new_shared(numel)
            out = elem.new(storage)

        if isinstance(_elem, th.Tensor):
            return th.stack(batch, 0, out=out)
        else:
            dim = _elem.dim
            if isinstance(_elem, CollateByCat):
                return th.cat(batch, dim=dim, out=out)
            elif isinstance(_elem, CollateByStack):
                return th.stack(batch, dim=dim, out=out)

    elif elem_type.__module__ == 'numpy' and elem_type.__name__ != 'str_' \
            and elem_type.__name__ != 'string_':
        elem = batch[0]
        if elem_type.__name__ == 'ndarray':
            # array of string classes and object
            if np_str_obj_array_pattern.search(elem.dtype.str) is not None:
                raise TypeError(default_collate_err_msg_format.format(elem.dtype))

            return default_collate([th.as_tensor(b) for b in batch])
        elif elem.shape == ():  # scalars
            return th.as_tensor(batch)
    elif isinstance(elem, float):
        return th.tensor(batch, dtype=th.float64)
    elif isinstance(elem, int_classes):
        return th.tensor(batch)
    elif isinstance(elem, string_classes):
        return batch
    elif isinstance(elem, container_abcs.Mapping):
        # checks individual field within dictionary
        return {key: default_collate([d[key] for d in batch]) for key in elem}
    elif isinstance(elem, tuple) and hasattr(elem, '_fields'):  # namedtuple
        return elem_type(*(default_collate(samples) for samples in zip(*batch)))
    elif isinstance(elem, container_abcs.Sequence):
        # check to make sure that the elements in batch have consistent size
        it = iter(batch)
        _elem = len(next(it))
        elem_size = (_elem)
        if not all(len(elem) == elem_size for elem in it):
            raise RuntimeError('each element in list of batch should be of equal size')
        transposed = zip(*batch)
        return [default_collate(samples) for samples in transposed]
    elif isinstance(elem, DGLGraph):
        mods = elem.__class__.__module__.split('.')
        # cls_ = elem.__class__.__name__
        if 'compas_torch_fofin' in mods:
            mod = '.'.join(mods)  
            return elem.__class__.batch(batch)
        else:
            return dgl.batch(batch)
    # elif isinstance(elem, Uncollated):
    #     return [elem.item for elem in batch]
    elif elem is None:
        return batch
    else:
        return batch
        print(TypeError(default_collate_err_msg_format.format(elem_type)))

    def postprocess_batch(batch):
    """
    Post-processes a batch of data by transforming certain elements.

    Args:
        batch (dict): A dictionary where keys are batch item identifiers and values are lists of batch elements.

    Returns:
        dict: A new dictionary with processed batch elements.
    """
    batch_ = {}
    n = len(batch['i'])
    for _k, _v in batch.items():
        if isinstance(_v, list) and isinstance(_v[0], UncollatedWrapper):
            _v = [__v.item for __v in _v]
        elif _v is None or _v == [None]:
            _v = [''] * n
        batch_[_k] = _v
    return batch_
