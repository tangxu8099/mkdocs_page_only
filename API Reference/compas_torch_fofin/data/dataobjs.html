    class DataObject:
    """A class to represent a data object with various attributes and methods for data manipulation."""
    def __init__(self,
                 name: str = None,
                 domain: DomainBase = None,
                 dim: int = 1,
                 default: Any = None,
                 unit: Optional[str] = None,
                 dtype: Optional[str] = None,
                 transformations: Optional[List[str]] = None):
        """
        Initialize a new instance of the class.

        Args:
            name (str, optional): The name of the data object. Defaults to None.
            domain (DomainBase, optional): The domain associated with the data object. Defaults to None.
            dim (int, optional): The dimensionality of the data object. Defaults to 1.
            default (Any, optional): The default value for the data object. Defaults to None.
            unit (str, optional): The unit of measurement for the data object. Defaults to None.
            dtype (str, optional): The data type of the data object. Defaults to None.
            transformations (List[str], optional): A list of transformations to apply to the data object. Defaults to None.
        """

        self.name = name
        self.domain = domain
        self.dim = dim
        self._default = default
        self.unit = unit
        self.dtype = dtype
        self.transformations = transformations

    def default(self, device='cpu'):
        """
        Returns the default data cast to the specified device and data type.

        Args:
            device (str, optional): The device to cast the data to. Default is 'cpu'.

        Returns:
            The default data cast to the specified device and data type.
        """
        return _cast_data(self._default, self.dtype, device=device)

    @property
    def transformations(self):
        """
        Returns the transformations associated with the object.

        Returns:
            list: A list of transformation matrices.
        """
        return self._transformations

    @transformations.setter
    def transformations(self, transforms):
        """
        Apply a series of transformations to the object.

        Args:
            transforms (list or object): A list of transformation objects or a single transformation object.
        """
        _transforms = item_to_list(transforms or [])
        _transforms = [_t(dobj=self) if isinstance(_t, type) else _t for _t in _transforms]
        self._transformations = _transforms

    @property
    def has_default(self):
        """
        Check if the object has a default value.

        Returns:
            bool: True if the object has a default value, False otherwise.
        """
        return self._default is not None

    def random_samples(self, size: int):
        """
        Generate random samples within the domain.

        Args:
            size (int): The number of random samples to generate.

        Returns:
            numpy.ndarray: A 2D array of shape (size, self.dim) containing the random samples.
        """
        return self.domain.random_samples((size, self.dim))

    def grid_samples(self, size: int):
        """
        Generate grid samples within the domain.

        Args:
            size (int): The number of grid samples to generate.

        Returns:
            numpy.ndarray: A 2D array of shape (size, self.dim) containing the grid samples.
        """
        return self.domain.grid_samples((size, self.dim))

    def sample_around(self, centroid, size: int, percentage: int):
        """
        Samples points around a given centroid within a specified percentage of the domain.

        Args:
            centroid (array-like): The central point around which to sample.
            size (int): The number of points to sample.
            percentage (int): The percentage of the domain within which to sample around the centroid.

        Returns:
            array-like: A collection of sampled points around the centroid.
        """
        return self.domain.sample_around(centroid, (size, self.dim), percentage)

    def transform(self, data):
        """
        Applies transformations to the data.

        Args:
            data: The data to transform.

        Returns:
            The transformed data.
        """
        for tfr in self.transformations:
            data = tfr(data)
        return data

    def inverse_transform(self, data):
        """
        Applies inverse transformations to the data.

        Args:
            data: The data to inverse transform.

        Returns:
            The inverse transformed data.
        """
        for tfr in self.transformations:
            data = tfr.inverse_transform(data)  # Assuming transform is a callable
        return data

    def __str__(self):
        """
        Returns a string representation of the data object.

        Returns:
            str: The name of the data object.
        """
        return self.name
    def __repr__(self):
        """
        Returns a detailed string representation of the data object.

        Returns:
            str: A detailed string representation of the data object.
        """
        return f"{self.__class__.__name__}(name={self.name}, dim={self.dim}, domain={self.domain}, unit={self.unit}, dtype={self.dtype})"
