    class DataObjectTransform:
    """
    Base class for transforming PyTorch Tensors.

    To create a custom transformation, subclass this and implement:
    - `transform`: Transform the data
    - `inverse_transform`: Reverse the transformation
    - `_fit_partial`: Fit the transformation
    - `is_fitted`: Check if the transformation has been fitted

    If no fitting is required, set `requires_fitting` to `False`, and you don't need to implement `_fit_partial` or `is_fitted`.
    """

    name: str = None  # set by a decorator
    requires_fitting: bool = True
    _STATS = []

    def __init__(self, **kwargs):
        """
        Initializes the DataObjectTransform.

        Args:
            **kwargs: Additional keyword arguments.
        """
        dobj = kwargs.pop('dobj', None)
        if dobj is not None:
            self.fit_from_dataobject(dobj=dobj)

    def fit_from_dataobject(self, dobj: Optional[object]):
        """
        Fits the transformation from a data object.

        Args:
            dobj (Optional[object]): The data object to fit from.
        """
        [setattr(self, _k, getattr(dobj, _k)) for _k in self._STATS]

    def fit(self, X: th.Tensor) -> "DataObjectTransform":
        """
        Fits the transformation.

        Args:
            X (th.Tensor): The tensor to fit.

        Returns:
            DataObjectTransform: The fitted transformation.
        """
        pass

    def is_fitted(self) -> bool:
        """
        Checks if the transformation is fitted.

        Returns:
            bool: True if the transformation is fitted, False otherwise.
        """
        if self.requires_fitting:
            raise NotImplementedError()
        return True

    def reset(self):
        """
        Resets the transformation.

        Returns:
            DataObjectTransform: The reset transformation.
        """
        return self

    def fit_transform(self, X: th.Tensor, dobj: Optional[object] = None) -> th.Tensor:
        """
        Fits and transforms the data.

        Args:
            X (th.Tensor): The tensor to fit and transform.
            dobj (Optional[object]): The data object to fit from.

        Returns:
            th.Tensor: The transformed tensor.
        """
        return self.fit(X, dobj).transform(X, dobj)

    def transform(self, X: th.Tensor, dobj: Optional[object] = None) -> th.Tensor:
        """
        Transforms the data.

        Args:
            X (th.Tensor): The tensor to transform.
            dobj (Optional[object]): The data object to use for transformation.

        Returns:
            th.Tensor: The transformed tensor.
        """
        raise NotImplementedError()

    def inverse_transform(self, X: th.Tensor, dobj: Optional[object] = None) -> th.Tensor:
        """
        Inversely transforms the data.

        Args:
            X (th.Tensor): The tensor to inverse transform.
            dobj (Optional[object]): The data object to use for inverse transformation.

        Returns:
            th.Tensor: The inversely transformed tensor.
        """
        raise NotImplementedError()

    def copy(self, reset: bool = False):
        """
        Returns a copy of the transformation, optionally reset.

        Args:
            reset (bool): Whether to reset the transformation.

        Returns:
            DataObjectTransform: The copied transformation.
        """
        obj_copy = copy.copy(self)
        if reset:
            obj_copy.reset()
        return obj_copy

    def __jsondump__(self):

    def __jsonload__(cls, data):

        Args:
            data (dict): The JSON data.

        Returns:
            DataObjectTransform: The loaded transformation.

    class MinMaxScaler(DataObjectTransform):
    """
    Implements a min-max scaling of the data using PyTorch.
    """

    @property
    def _STATS(self):
        """
        Returns a list of statistical measures.

        Returns:
            list: A list containing the strings 'max' and 'min'.
        """
        return ['max', 'min']

    def __init__(self, target_range: tuple = (0, 1), dim: int = 0, **kwargs):
        """
        Initializes the MinMaxScaler.

        Args:
            target_range (tuple): The target range for the scaling.
            dim (int): The dimension to perform normalization on.
            **kwargs: Additional keyword arguments.
        """
        self.target_range = target_range
        self.dim = dim
        self.min = None
        self.max = None
        super().__init__(**kwargs)

    def is_fitted(self) -> bool:
        """
        Checks if the scaler is fitted.

        Returns:
            bool: True if the scaler is fitted, False otherwise.
        """
        return self.min is not None and self.max is not None

    def reset(self):
        """
        Resets the scaler.

        Returns:
            MinMaxScaler: The reset scaler.
        """
        self.min, self.max = None, None

    def fit(self, X: th.Tensor):
        """
        Fits the scaler to the data.

        Args:
            X (th.Tensor): The tensor to fit.
        """
        self.min = th.min(X, dim=self.dim).values
        self.max = th.max(X, dim=self.dim).values

    def __call__(self, data):
        """
        Calls the transform method.

        Args:
            data (th.Tensor): The tensor to transform.

        Returns:
            th.Tensor: The transformed tensor.
        """
        return self.transform(data)

    def transform(self, X: th.Tensor) -> th.Tensor:
        """
        Transforms the data.

        Args:
            X (th.Tensor): The tensor to transform.

        Returns:
            th.Tensor: The transformed tensor.
        """
        if not self.is_fitted():
            raise ValueError("MinMaxScaler has not been fitted yet.")

        centered = X - self.min.to(device=X.device)
        span = (self.max - self.min).to(device=X.device)

        if th.allclose(span, th.zeros_like(span)):  # span is zero if all values are the same
            X = centered  # When all values are the same, just return centered values (all zero)
        else:
            X = centered / span  # Normalize to 0-1 range

        return X * (self.target_range[1] - self.target_range[0]) + self.target_range[0]  # Scale to target range

    def inverse_transform(self, X: th.Tensor) -> th.Tensor:
        """
        Inversely transforms the data.

        Args:
            X (th.Tensor): The tensor to inverse transform.

        Returns:
            th.Tensor: The inversely transformed tensor.
        """
        if not self.is_fitted():
            raise ValueError("MinMaxScaler has not been fitted yet.")

        X = (X - self.target_range[0]) / (self.target_range[1] - self.target_range[0])  # Scale to 0-1 range
        return X * (self.max - self.min) + self.min  # Scale back to original range

    def inverse_transform(self, X: th.Tensor) -> th.Tensor:
    """
    Inversely transforms the data.

        X (th.Tensor): The tensor to inverse transform.

    Returns:
        th.Tensor: The inversely transformed tensor.

    if not self.is_fitted():
        raise ValueError("MinMaxScaler has not been fitted yet.")

    X = (X - self.target_range[0]) / (self.target_range[1] - self.target_range[0])  # Scale to 0-1 range
    return X * (self.max - self.min) + self.min  # Scale back to original range

    def transform(self, X: th.Tensor) -> th.Tensor:
    """
    Transforms the data.

    Args:
        X (th.Tensor): The tensor to transform.

    Returns:
        th.Tensor: The transformed tensor.
    """
    if not self.is_fitted():
        raise ValueError("MinMaxScaler has not been fitted yet.")

    centered = X - self.min.to(device=X.device)
    span = (self.max - self.min).to(device=X.device)

    if th.allclose(span, th.zeros_like(span)):  # span is zero if all values are the same
        X = centered  # When all values are the same, just return centered values (all zero)
    else:
        X = centered / span  # Normalize to 0-1 range

    return X * (self.target_range[1] - self.target_range[0]) + self.target_range[0]  # Scale to target range

    class Standardization(DataObjectTransform):
    """Implements standardization as (x - mean) / std using PyTorch."""

    def __init__(self, dim: int = 0):
        """
        Initializes the Standardization.

        Args:
            dim (int): The dimension to perform standardization on.
        """
        super().__init__()
        self.mean = None
        self.std = None
        self.dim = dim

    def is_fitted(self) -> bool:
        """
        Checks if the standardization is fitted.

        Returns:
            bool: True if the standardization is fitted, False otherwise.
        """
        return self.mean is not None and self.std is not None

    def reset(self):
        """
        Resets the mean and std values.

        Returns:
            Standardization: The reset standardization.
        """
        self.mean, self.std = None, None

    def _fit_partial(self, X: th.Tensor, dobj: Optional[object] = None) -> None:
        """
        Fits the mean and std from the data.

        Args:
            X (th.Tensor): The tensor to fit.
            dobj (Optional[object]): The data object to fit from.
        """
        self.mean = th.mean(X, dim=self.dim, keepdim=True)
        self.std = th.std(X, dim=self.dim, keepdim=True)

    def transform(self, X: th.Tensor, dobj: Optional[object] = None) -> th.Tensor:
        """
        Standardizes the data (X - mean) / std.

        Args:
            X (th.Tensor): The tensor to transform.
            dobj (Optional[object]): The data object to use for transformation.

        Returns:
            th.Tensor: The standardized tensor.
        """
        if not self.is_fitted():
            raise ValueError("The transform has not been fitted yet.")
        return (X - self.mean) / (self.std + CFG.get('TOLERANCE.NUMERICAL'))

    def inverse_transform(self, X: th.Tensor, dobj: Optional[object] = None) -> th.Tensor:
        """
        Reverts the standardized data to the original scale.

        Args:
            X (th.Tensor): The tensor to inverse transform.
            dobj (Optional[object]): The data object to use for inverse transformation.

        Returns:
            th.Tensor: The inversely transformed tensor.
        """
        if not self.is_fitted():
            raise ValueError("The transform has not been fitted yet.")
        return X * self.std + self.mean
