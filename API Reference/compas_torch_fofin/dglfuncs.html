    class DGLFunctions(object):
    """
    DGLFunctions is a collection of static and instance methods for manipulating and processing graph data,
    particularly for use with the Deep Graph Library (DGL). These methods include operations on edges and nodes,
    such as copying features, concatenating features, performing matrix multiplications, and computing various statistics.
    """
    # =============================================================================
    # =============================================================================
    # Edge Funcs
    # =============================================================================
    # =============================================================================

    def copy_feats(feats, out=None, _attr='data'):
        """
        Creates a function that copies features from the input to the output.

        Args:
            feats_link (str or list of str): The feature(s) to be copied. If a string is provided, it will be converted to a list.
            feats_out (str or list of str, optional): The output feature name(s). If a string is provided, it will be used as a prefix for the input features. Defaults to 'mail_'.

        Returns:
            function: A function that takes edges as input and returns a dictionary mapping the output feature names to the corresponding edge data.
        """
        feats   = feat_to_feats(feats)
        out     = name_messages(out, feats)

        def inner(links):
            return {_out: getattr(links, _attr)[_feat] for _feat, _out in zip(feats, out)}
        return inner

    copy_us     = partialmethod(copy_feats, _attr='src')
    copy_vs     = partialmethod(copy_feats, _attr='dst')
    copy_es     = partialmethod(copy_feats, _attr='data')

    # @staticmethod
    def copy_combined_feats(feats, out='mail', reduce_='cat', _attr='data'):   # PREVIOUS NAME copy_es
        """
        Creates a function that concatenates specified edge features.

        Args:
            feats_link (list of str): List of edge feature names to concatenate.
            out (str, optional): The key under which the concatenated data will be stored. Defaults to 'mail_'.

        Returns:
            function: A function that takes edges as input and returns a dictionary with the concatenated edge features.
        """
        feats   = feat_to_feats(feats)
        def inner(links):
           data = th.cat([getattr(links, _attr) for _feat in feats], dim=1)
           return {out: data}
        return inner

    copy_cat_us     = partialmethod(copy_feats, _attr='src')
    copy_cat_vs     = partialmethod(copy_feats, _attr='dst')
    copy_cat_s      = partialmethod(copy_feats, _attr='data')

    def copy(feats_src=[], feats_dst=[], feats_link=[]):
        """
        Creates a function that copies specified features from source, destination, and link edges.

        Args:
            feats_src (list, optional): List of feature names to copy from the source edges. Defaults to an empty list.
            feats_dst (list, optional): List of feature names to copy from the destination edges. Defaults to an empty list.
            feats_link (list, optional): List of feature names to copy from the link edges. Defaults to an empty list.

        Returns:
            function: A function that takes edges as input and returns a dictionary with the copied features.
        """
        feats_src, feats_dst, feats_link = [feat_to_feats(_feats) 
                                            for _feats in [feats_src, feats_dst, feats_link]]
        def inner(edges):
            res = {}
            res.update({_f: edges.src[_f] for _f in feats_src})
            res.update({_f: edges.dst[_f] for _f in feats_dst})
            res.update({_f: edges.data[_f] for _f in feats_link})
            return res
        return inner

    def copy_u_and_e(feat_src, feat_link, feat_out_src='mail', feat_out_link='w'):
        """
        Creates a function that copies specified features from source nodes and edges.

        Args:
            feat_src (str): The feature name to copy from the source nodes.
            feat_link (str): The feature name to copy from the edges.
            feat_out_src (str, optional): The output feature name for the source nodes. Defaults to '_m'.
            feat_out_link (str, optional): The output feature name for the edges. Defaults to '_w'.

        Returns:
            function: A function that takes edges as input and returns a dictionary with the copied features.
        """
        def inner(edges):
            return {feat_out_src: edges.src[feat_src],
                    feat_out_link: edges.data[feat_link]}
        return inner

    def u_matmul_e(nfeat, efeat, out='_m'):
        """
        Multiplies node features with edge features using matrix multiplication.

        Args:
            nfeat (str): The key for node features in the graph data.
            efeat (str): The key for edge features in the graph data.
            out (str, optional): The key for the output result. Defaults to '_m'.

        Returns:
            function: A function that takes edges as input and returns a dictionary
                      with the result of the matrix multiplication.
        """
        def inner(edges):
            res = th.matmul(edges.src[nfeat].unsqueeze(1), edges.data[efeat])
            return {out: res}

    def e_mul_e(efeat_1, efeat_2, out='mail'):
        """
        Element-wise multiplication of edge features.

        This function returns an inner function that performs element-wise 
        multiplication of two edge features and stores the result in a new 
        edge feature.

        Args:
            efeat_1 (str): The key for the first edge feature.
            efeat_2 (str): The key for the second edge feature.
            out (str, optional): The key for the output edge feature. Defaults to '_m'.

        Returns:
            function: A function that takes edges as input and returns a dictionary 
            with the result of the element-wise multiplication stored under the key `out`.
        """
        def inner(edges):
            res = edges.data[efeat_1] * edges.data[efeat_2]
            return {out: res}
        return inner

    def e_prod(efeats, out='mail'):
        """
        Element-wise multiplication of edge features.

        This function returns an inner function that performs element-wise 
        multiplication of two edge features and stores the result in a new 
        edge feature.

        Args:
            efeat_1 (str): The key for the first edge feature.
            efeat_2 (str): The key for the second edge feature.
            out (str, optional): The key for the output edge feature. Defaults to '_m'.

        Returns:
            function: A function that takes edges as input and returns a dictionary 
            with the result of the element-wise multiplication stored under the key `out`.
        """
        def inner(edges):
            res = reduce(th.mul, [edges.data[_feat] for _feat in efeats])
            return {out: res}
        return inner

    def e_norm(efeat, out=None):
        """
        Creates a function that computes the Euclidean norm of edge features.

        Args:
            efeat (str): The key for the edge features in the edge data.
            out (str, optional): The key for the output norm in the edge data. If not provided, defaults to 'mail_' + efeat.

        Returns:
            function: A function that takes an edge data object and returns a dictionary with the computed norm.
        """
        def inner(edges):
            res = th.linalg.norm(edges.data[efeat], dim=-1, keepdim=True)
            out_ = out or 'mail_' + efeat
            return {out_: res}
        return inner

    def e_normalize(efeat, out=None):
        """
        Creates a function that computes the Euclidean norm of edge features.

        Args:
            efeat (str): The key for the edge features in the edge data.
            out (str, optional): The key for the output norm in the edge data. If not provided, defaults to 'mail_' + efeat.

        Returns:
            function: A function that takes an edge data object and returns a dictionary with the computed norm.
        """
        def inner(edges):
            res = th.linalg.normalize(edges.data[efeat], dim=-1, keepdim=True)
            out_ = out or 'mail_' + efeat
            return {out_: res}
        return inner

    # @staticmethod
    def u_sub_v_squared(feat_src, feat_dst=None, feat_out='mail'):
        """
        Computes the squared difference between source and destination features.

        This function returns another function that calculates the squared difference
        between the source and destination features of edges in a graph.

        Args:
            feat_src (str): The key for the source feature.
            feat_dst (str, optional): The key for the destination feature. Defaults to None.
            feat_out (str, optional): The key for the output feature. Defaults to '_m'.

        Returns:
            function: A function that takes edges as input and returns a dictionary with the 
                      squared difference between the source and destination features.
        """
        def inner(edges):
            res = (edges.src[feat_src] - edges.dst[feat_dst])**2
            return {feat_out: res}
        return inner

    def identity_n(msg, out=None, type_='n'):
        """
        Creates an identity function for node messages in a graph neural network.

        Args:
            msg (str): The key for the message to be used from the mailbox.
            out (str, optional): The key for the output message. Defaults to None.
            type_ (str, optional): The type of the message, defaults to 'n'.

        Returns:
            function: A function that takes nodes as input and returns a dictionary 
            with the output message.
        """
        msg = msg or out
        def inner(nodes):
            return {out: nodes.mailbox[msg]}
        return inner

    def weighted_mean(feat, w, out, tol=1e-6):
        """
        Compute the weighted mean of node features.
        This function computes the weighted mean of node features using the
        specified feature and weight attributes from the nodes' mailbox.

        Args:
            feat (str): The feature attribute name in the nodes' mailbox.
            w (str): The weight attribute name in the nodes' mailbox.
            out (str): The output attribute name for the computed weighted mean.
            tol (float, optional): A tolerance value to avoid division by zero. Defaults to 1e-6.

        Returns:
            function: A function that computes the weighted mean for the given nodes.
        """
        def inner(nodes):
            # n_n x deg x d
            X_wd        = nodes.mailbox[feat] * nodes.mailbox[w].unsqueeze(dim=-1)
            # n_n x 1 x d
            X_wd_sum    = nodes.mailbox[w].sum(dim=1, keepdim=True).unsqueeze(-1)

            X_wd_mean   = th.zeros_like(X_wd)
            mask        = X_wd_sum.flatten() >= th.tensor(CONFIG.get('TOLERANCE.NUMERICAL'), device=X_wd.device)

            if mask.sum() > 0.:
                X_wd_mean[mask] = X_wd[mask] / X_wd_sum[mask]

            X_wd_mean = X_wd_mean.sum(dim=1)

            return {out: X_wd_mean}
        return inner

    def masked_reduce(feat, mask, out, reduce='mean'):
        """
        Applies a masked reduction operation on node features within a graph.
        This function performs a reduction operation (e.g., mean, sum) on the node features
        of a graph, considering only the elements specified by a mask. The reduction is 
        applied to the mailbox of the nodes, which contains messages from neighboring nodes.

        Args:
            feat (str): The feature key in the mailbox to be reduced.
            mask (str): The mask key in the mailbox to specify which elements to consider.
            out (str): The key under which the reduced result will be stored.
            reduce (str or callable, optional): The reduction operation to apply. 
                Can be a string representing a method name (e.g., 'mean', 'sum') or a callable 
                that performs the reduction. Default is 'mean'.

        Returns:
            function: A function that takes nodes as input and returns a dictionary with the 
            reduced features stored under the specified key.
        """

        def inner(nodes):
            # n_n x deg x d
            X = nodes.mailbox[feat]
            # n_n x deg x 1
            X_msk = th.ones(X.shape[:-1] + (1,)).to(th.bool).to(X.device)
            if mask is not None:
                X_msk = nodes.mailbox[mask].to(th.bool)

            # n_nodes x 1
            X_msk_sum = X_msk.sum(dim=1).squeeze(-1)

            # [] of deg x d
            X_ = X[X_msk]  # flatten into tensor of shape [unmasked elements x 1]
            X_ = X_.split(X_msk_sum.tolist(), dim=0)

            # [] of d
            X_rdc = []

            for _X in X_:
                if _X.numel() == 0:
                    _X_rdc = th.tensor([th.float('nan')])
                else:
                    if isinstance(reduce, str):
                        _X_rdc = getattr(_X, reduce)(dim=0)
                    else:
                        _X_rdc = reduce(_X)(dim=0)
                X_rdc.append(_X_rdc) 

            X_rdc = th.stack(X_rdc, dim=0) 
            return {out: X_rdc}
        return inner

    def mass_add(feats, out):
        """
        Creates a function that aggregates specified features from nodes' mailbox and data.

        Args:
            feats (Union[str, List[str]]): A feature or list of features to be aggregated.
            out (str): The key under which the aggregated result will be stored.

        Returns:
            Callable: A function that takes nodes as input and returns a dictionary with the aggregated feature.
        """
        feats = [feats] if isinstance(feats, str) else feats

        def inner(nodes):
            feats_mb = [_f for _f in feats if _f in (nodes.mailbox or [])]
            feats_dat = [_f for _f in feats if _f in (nodes.data or [])]

            Xs_mb = [nodes.mailbox[_feat] for _feat in feats_mb]
            Xs_dat = [nodes.data[_feat] for _feat in feats_dat]
            Xs_all = Xs_mb + Xs_dat
            assert Xs_all

            X_stk = th.stack(Xs_all, dim=-2)
            X_stk_sum = th.sum(X_stk, dim=-2)

            return {out: X_stk_sum}

        return inner

    def mass_reduce(feats, out, reduce_='sum'):

            res         = reduce_tensor_list(Xs_all, reduce_=reduce_)

    def masked_reduce(feat, mask, out, reduce='mean'):
    """
    Applies a masked reduction operation on node features within a graph.
    This function performs a reduction operation (e.g., mean, sum) on the node features
    of a graph, considering only the elements specified by a mask. The reduction is 
    applied to the mailbox of the nodes, which contains messages from neighboring nodes.

    Args:
        feat (str): The feature key in the mailbox to be reduced.
        mask (str): The mask key in the mailbox to specify which elements to consider.
        out (str): The key under which the reduced result will be stored.
        reduce (str or callable, optional): The reduction operation to apply. 
            Can be a string representing a method name (e.g., 'mean', 'sum') or a callable 
            that performs the reduction. Default is 'mean'.

    Returns:
        function: A function that takes nodes as input and returns a dictionary with the 
        reduced features stored under the specified key.
    """

    def inner(nodes):
        # n_n x deg x d
        X = nodes.mailbox[feat]
        # n_n x deg x 1
        X_msk = th.ones(X.shape[:-1] + (1,)).to(th.bool).to(X.device)
        if mask is not None:
            X_msk = nodes.mailbox[mask].to(th.bool)

        # n_nodes x 1
        X_msk_sum = X_msk.sum(dim=1).squeeze(-1)

        # [] of deg x d
        X_ = X[X_msk]  # flatten into tensor of shape [unmasked elements x 1]
        X_ = X_.split(X_msk_sum.tolist(), dim=0)

        # [] of d
        X_rdc = []

        for _X in X_:
            if _X.numel() == 0:
                _X_rdc = th.tensor([th.float('nan')])
            else:
                if isinstance(reduce, str):
                    _X_rdc = getattr(_X, reduce)(dim=0)
                else:
                    _X_rdc = reduce(_X)(dim=0)
            X_rdc.append(_X_rdc) 

        X_rdc = th.stack(X_rdc, dim=0) 
        return {out: X_rdc}
    return inner

    def weighted_mean(feat, w, out, tol=1e-6):
    """
    Compute the weighted mean of node features.
    This function computes the weighted mean of node features using the
    specified feature and weight attributes from the nodes' mailbox.

    Args:
        feat (str): The feature attribute name in the nodes' mailbox.
        w (str): The weight attribute name in the nodes' mailbox.
        out (str): The output attribute name for the computed weighted mean.
        tol (float, optional): A tolerance value to avoid division by zero. Defaults to 1e-6.

    Returns:
        function: A function that computes the weighted mean for the given nodes.
    """
    def inner(nodes):
        # n_n x deg x d
        X_wd        = nodes.mailbox[feat] * nodes.mailbox[w].unsqueeze(dim=-1)
        # n_n x 1 x d
        X_wd_sum    = nodes.mailbox[w].sum(dim=1, keepdim=True).unsqueeze(-1)

        X_wd_mean   = th.zeros_like(X_wd)
        mask        = X_wd_sum.flatten() >= th.tensor(CONFIG.get('TOLERANCE.NUMERICAL'), device=X_wd.device)

        if mask.sum() > 0.:
            X_wd_mean[mask] = X_wd[mask] / X_wd_sum[mask]

        X_wd_mean = X_wd_mean.sum(dim=1)

        return {out: X_wd_mean}
    return inner
