    class Graph(CGraph):
    """A class representing a graph, extending the COMPAS Graph class."""

    def edges_in_node_set(self, nkeys, ekeys=None):
        """
        Get edges within a set of nodes.

        Args:
            nkeys (list): List of node keys.
            ekeys (list, optional): List of edge keys. Defaults to None.

        Returns:
            list: List of edges within the node set.
        """
        ekeys = ekeys or list(sorted(self.edges()))
        eks = [_ek for _ek in ekeys if _ek[0] in nkeys and _ek[1] in nkeys]
        return eks

    def to_subgraph(self, nks, eks, nattrs, eattrs, reorder_keys=True):
        """
        Create a subgraph from the current graph.

        Args:
            nks (list): List of node keys for the subgraph.
            eks (list): List of edge keys for the subgraph.
            nattrs (dict): Node attributes.
            eattrs (dict): Edge attributes.
            reorder_keys (bool, optional): Whether to reorder keys. Defaults to True.

        Returns:
            Graph: The subgraph.
        """
        from copy import deepcopy

        subg = self.__class__()

        if reorder_keys:
            nk0_to_nk1 = dict(zip(nks, range(len(nks))))
            ek0_to_ek1 = {tuple(sorted((_u, _v))): tuple(sorted((nk0_to_nk1[_u], nk0_to_nk1[_v])))
                          for _u, _v in eks}
        else:
            nk0_to_nk1 = dict(zip(nks, nks))
            ek0_to_ek1 = dict(zip(eks, eks))

        for _nk_0, _nk_1 in nk0_to_nk1.items():
            _nattrs = deepcopy(self.node[_nk_0])
            subg.add_node(_nk_1, **_nattrs)

        for _ek_0, _ek_1 in ek0_to_ek1.items():
            _u_0, _v_0 = _ek_0
            _eattrs = deepcopy(self.edge.get(_u_0, {}).get(_v_0, {}))
            subg.add_edge(_u_0, _v_0, attr_dict=_eattrs)

        subg.attributes = self.attributes
        subg.default_node_attributes = self.default_node_attributes
        subg.default_edge_attributes = self.default_edge_attributes

        return subg

    def node_neighbors(self, key):
        """
        Get neighbors of a node.

        Args:
            key: Node key.

        Returns:
            list: List of neighboring nodes.
        """
        return self.neighbors(key)

    def edge_neighbors(self, ekey, nkey=None):
        """
        Get neighbors of an edge.

        Args:
            ekey (tuple): Edge key.
            nkey (optional): Node key. Defaults to None.

        Returns:
            list or dict: List of neighbors if nkey is provided, otherwise a dictionary of neighbors.
        """
        nks_to_chk = ekey
        if nkey:  # get lower neighbors connected to one vertex only
            assert nkey in nks_to_chk
            nks_to_chk = [nkey]

        lb_nbrs = {_nk: [(_nbr, _nk)
                         for _nbr in self.node_neighbors(_nk)
                         if _nbr not in ekey]
                   for _nk in nks_to_chk}
        if nkey:
            return list(lb_nbrs.values())[0]
        return lb_nbrs

    @property
    def nks(self):
        """
        List of node keys.

        Returns:
            list: List of node keys.
        """
        return list(self.nodes())

    @property
    def eks(self):
        """
        List of edge keys.

        Returns:
            list: List of edge keys.
        """
        return list(self.edges())

    @property
    def eks_r(self):
        """
        List of edge keys including reversed edges.

        Returns:
            list: List of edge keys including reversed edges.
        """
        eks = list(self.edges())
        eks += list(map(lambda x: x[::-1], eks))
        return eks

    @property
    def ek_to_ei(self):
        """
        Mapping of edge keys to edge indices.

        Returns:
            dict: Dictionary mapping edge keys to edge indices.
        """
        eks = self.eks
        return dict(zip(eks, range(len(eks))))

    @property
    def ek_to_ei_r(self):
        """
        Mapping of edge keys to edge indices including reversed edges.

        Returns:
            dict: Dictionary mapping edge keys to edge indices including reversed edges.
        """
        ek_to_ei = self.ek_to_ei
        ret = {**ek_to_ei,
               **{_evk[::-1]: _ei for _evk, _ei in ek_to_ei.items()}}
        return ret
