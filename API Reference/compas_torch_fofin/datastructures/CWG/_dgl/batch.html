    class CWGBatching:
    """Class for batching and unbatching CWG objects."""

    @property
    def is_batched(self):
        """Check if the CWG is batched.

        Returns:
            bool: True if the batch size is greater than 1, False otherwise.
        """
        return self.batch_size > 1

    @classmethod
    def batch(cls, CWGs, copy=True, enforce_common_schema=True):
        """Batch multiple CWG objects into a single batched CWG.

        Args:
            CWGs (list): List of CWG objects to be batched.
            copy (bool, optional): Whether to copy the CWG objects. Defaults to True.
            enforce_common_schema (bool, optional): Whether to enforce a common schema. Defaults to True.

        Returns:
            CWGBatching: A new batched CWG object.
        """
        cls = CWGs[0].__class__
        cls_name = cls.__name__
        assert all([_CWG.__class__.__name__ == cls_name for _CWG in CWGs])

        CWGs = [_CWG.local_var() for _CWG in CWGs]
        CWGs = cls.enforce_common_schema(CWGs) if enforce_common_schema else CWGs

        n_nodes = {_ntype: 0 for _ntype in cls._NTYPES}
        for _i, _CWG in enumerate(CWGs):
            for _dim_cwc, _ltypes in cls._LTYPES.items():
                for _ltype in _ltypes:
                    _ntype_src, _ntype_link, _ntype_dst = _ltype
                    _is_link = _ntype_link != ''
                    # ensure out of place operation to accommodate graphs with local_var()
                    if _is_link and _CWG.edges(etype=_ltype)[0].numel():
                        _CWG.edges[_ltype].data['lid'] = _CWG.edges[_ltype].data['lid'] + n_nodes[_ntype_link]
            for _ntype, _n in n_nodes.items():
                n_nodes[_ntype] += _CWG.number_of_nodes(_ntype)

        bCWG = dgl.batch(CWGs)
        bCWG = cls(CWG=bCWG)
        bCWG.batch_num_elems = {_ntype: th.tensor([_CWG.number_of_nodes(_ntype) for _CWG in CWGs])
                                for _ntype in cls._NTYPES}
        bCWG._sttgs = CWGs[0]._sttgs
        return bCWG

    @property
    def batch_num(self):
        """Get the batch number.

        Returns:
            int: The number of 'g' nodes in the batch.
        """
        return self.number_of_('g')

    def unbatch(self):
        """Unbatch a batched CWG into individual CWGs.

        Returns:
            list: List of individual CWG objects.
        """
        CWGs = dgl.unbatch(self)
        CWGs = [self.__class__(CWG=_CWG) for _CWG in CWGs]
        n_nodes = {_ntype: 0 for _ntype in self._NTYPES}

        for _i, _CWG in enumerate(CWGs):
            for _dim_cwc, _ltypes in self._LTYPES.items():
                for _ltype in _ltypes:
                    _ntype_src, _ntype_link, _ntype_dst = _ltype
                    _is_link = _ntype_link != ''
                    if _is_link and _CWG.edges(etype=_ltype)[0].numel():
                        _CWG.edges[_ltype].data['lid'] = _CWG.edges[_ltype].data['lid'] - n_nodes[_ntype_link]
            for _ntype, _n in n_nodes.items():
                n_nodes[_ntype] += _CWG.number_of_nodes(_ntype)

        return CWGs

    @classmethod
    def enforce_common_schema(cls, CWGs):
        """Enforce a common schema across multiple CWG objects.

        Args:
            CWGs (list): List of CWG objects.

        Returns:
            list: List of CWG objects with a common schema.
        """
        # TODO

        id_to_dtype_cpu         = {0: th.bool, 1: th.int32, 2: th.int64, 3: th.float32, 4: th.float64}
        # id_to_dtype_cuda        = {0: th.cuda.BoolTensor, 1: th.cuda.IntTensor, 2: th.cuda.LongTensor, 3: th.cuda.FloatTensor, 4: th.cuda.DoubleTensor}
        id_to_dtype_cuda        = {0: th.bool, 1: th.int32, 2: th.int64, 3: th.float32, 4: th.float64}
        device_to_id_to_dtype   = {th.device('cpu'): id_to_dtype_cpu, CWGs[0].device: id_to_dtype_cuda}

        dtype_to_id = {}
        dtype_to_id.update({_v: _k for _k, _v in id_to_dtype_cpu.items()})
        dtype_to_id.update({_v: _k for _k, _v in id_to_dtype_cuda.items()})

        type_to_feats = {}

        # HACKY
        for _i, _CWG in enumerate(CWGs):
            _CWG.nodes['_g'].data.clear()

        for _i, _CWG in enumerate(CWGs):
            for _type, _feats in _CWG.feats_dict.items():
                _feats = set(list(_feats.keys()))
                if _i == 0:
                    type_to_feats.setdefault(_type, _feats)
                else:
                    type_to_feats[_type] = set.intersection(type_to_feats[_type], _feats)

        for _CWG in CWGs:
            for _type, _feats in _CWG.feats_dict.items():
                for _feat in _feats:
                    if _feat not in type_to_feats[_type]:
                        _CWG._feats_data[_type].pop(_feat)

        type_to_feat_to_dtype = {}
        for _type, _feats in type_to_feats.items():
            type_to_feat_to_dtype.setdefault(_type, {})
            for _feat in _feats:
                for _CWG in CWGs:
                    _val = _CWG.nodes[_type].data[_feat]
                    type_to_feat_to_dtype[_type].setdefault(_feat, [])
                    type_to_feat_to_dtype[_type][_feat].append(dtype_to_id[_val.dtype])
                type_to_feat_to_dtype[_type][_feat] = min(type_to_feat_to_dtype[_type][_feat])

        for _type, _feats in type_to_feats.items():
            for _feat in _feats:
                for _CWG in CWGs:
                    _val = _CWG.nodes[_type].data[_feat]
                    _dtype =  device_to_id_to_dtype[_CWG.device][type_to_feat_to_dtype[_type][_feat]]
                    if _val.dtype != _dtype:
                        _CWG._feats_data[_type][_feat] = _val.to(_dtype)

        return CWGs
