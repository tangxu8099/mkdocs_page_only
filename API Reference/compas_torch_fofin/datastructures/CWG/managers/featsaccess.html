    class CWGFeaturesAccess:
    """
    Class for accessing CWG features.

    Attributes:
        _REGISTERED_FEATS_FUNCS: Dictionary of registered feature functions.
        RECOMPUTE_FEATS: Flag indicating whether to recompute features.
    """

    @property   
    def nodes_data(self):
        """
        Get the data of the nodes.

        Returns:
            dict: The data of the nodes.
        """
        # gets only available data
        dct = {_elem: self.nodes[_elem].data for _elem in self.ntypes}
        return dct

    @property
    def feat_getset(self):
        """
        Get the feature get/set accessor.

        Returns:
            _CWGFeaturesGetSetAccessor: The feature get/set accessor.
        """
        return _CWGFeaturesGetSetAccessor(self, 0)

    @property
    def feats_getset(self):
        """
        Get the multiple features get/set accessor.

        Returns:
            _CWGFeaturesGetSetAccessor: The multiple features get/set accessor.
        """
        return _CWGFeaturesGetSetAccessor(self, 1)

    @property
    def feat(self):
        """
        Get the nodes features attributor.

        Returns:
            _CWGNodesFeaturesAttributor: The nodes features attributor.
        """
        return _CWGNodesFeaturesAttributor(self)

    # =============================================================================
    # Features
    # =============================================================================

    def _feat_needs_calc(self, type_, feat):
        """
        Check if a feature needs to be calculated.

        Args:
            type_ (str): The type of the feature.
            feat (str): The feature name.

        Returns:
            bool: True if the feature needs to be calculated, False otherwise.
        """
        if self.RECOMPUTE_FEATS is not None:
            return self.RECOMPUTE_FEATS
        has_feat_func = self.has_feat_func(type_, feat)
        if has_feat_func:
            feat_sttgs = self.feat_settings(type_, feat, None, raise_error=False)
            if feat_sttgs is not None:
                return feat_sttgs.recompute
        return False

    def _feat_can_be_init(self, type_, feat, default=None):
        """
        Check if a feature can be initialized.

        Args:
            type_ (str): The type of the feature.
            feat (str): The feature name.
            default: The default value for the feature.

        Returns:
            bool: True if the feature can be initialized, False otherwise.
        """
        feat_dsttgs = None

        init_feat = None
        if default is None:
            feat_sttgs = self.feat_settings(type_, feat)
            init_feat = feat_sttgs.init_feat if feat_sttgs else False
            feat_dsttgs = self.feat_data_settings(type_, feat, None, raise_error=False)
        elif isinstance(default, DataObject):
            init_feat = True
            feat_dsttgs = default

        if feat_dsttgs is not None:
            return init_feat and feat_dsttgs.has_default

        return None

    def has_feat_func(self, type_, feat):
        """
        Check if a feature function is registered.

        Args:
            type_ (str): The type of the feature.
            feat (str): The feature name.

        Returns:
            bool: True if the feature function is registered, False otherwise.
        """
        return feat in self._REGISTERED_FEATS_FUNCS[type_]

    # general elems feats accessor
    def elems_feat(self, feat, type_='v', value=None, 
                   idx=None, keys=None, default=None, recompute=None):
        """
        Access a feature of elements.

        Args:
            feat (str / DataObject): The feature name.
            type_ (str): The type of the elements.
            value: The value to set for the feature.
            idx: The indices of the elements.
            keys: The keys of the elements.
            default: The default value for the feature.
            recompute (bool): Whether to recompute the feature.

        Returns:
            The feature value.
        """

        n = self.number_of_(type_ if not type_.startswith('_') else type_[1:])
        _data = self.nodes[type_].data

        if keys is not None:
            idx = self.keys_to_indices(type_=type_, keys=keys)
        if idx is not None:
            idx = th.tensor(idx, device=self.device, dtype=th.long) \
                  if not isinstance(idx, th.Tensor) else idx

        # -----------------------------------------------------------------------------
        # Getter
        # -----------------------------------------------------------------------------

        if value is None:
            compute = recompute if recompute is not None else None

            # Try getting the feature
            value = _data.get(str(feat))

            # if features are not found:
            if value is None:
                compute = self.has_feat_func(type_, feat)
                init_feat = self._feat_can_be_init(type_, feat, default)
            # if features are found:
            else:
                compute = self._feat_needs_calc(type_, feat)
                init_feat = False

            if compute:
                func = self._REGISTERED_FEATS_FUNCS[type_][feat]
                func = getattr(self, func)
                value = func()

            if value is None and init_feat:
                value = self.feat_default(feat=feat, type_=type_, 
                                          default=default, repeat=True,
                                          raise_error=False)

            if value is None:
                raise KeyError("Cannot find '{}' feat: {}".format(type_, feat))
            else:
                if compute:
                    if func.__dict__.get('autoset', True) is True:
                        if type_ == '_g' and len(value.shape) >= 2 and value.shape[0] != 1:
                            _data[feat] = value.unsqueeze(0)
                        else:
                            _data[feat] = value
                    if func.__dict__.get('return_res', True) is False:
                        return _data[feat]

                else:
                    if type_ == '_g' and len(value.shape) >= 2 and value.shape[0] == 1:
                        value = value.squeeze(0)
                if idx is not None:
                    value = value[idx]
                return value

        # -----------------------------------------------------------------------------
        # Setter
        # -----------------------------------------------------------------------------

        else:
            if not isinstance(value, th.Tensor):
                value = th.tensor(value).to(self.device)

            n_ = idx.shape[0] if idx is not None else n
            d = len(value.shape)

            if value.numel() == 1 and n_ > 1:
                value = th.ones(n_, 1).to(self.device) * value

            if d <= 2:
                value = value.reshape(n_, -1)
            else:
                assert n_ == value.shape[0]
                pass

            # only allow instantiation if index provided? why this again?
            value_ = value
            if idx is not None:
                if feat not in _data:
                    value_ = th.zeros(n, value.shape[1]).to(value.dtype, value.device)
                value_[idx] = value

            if 'g' not in type_:
                self.update_node_data(ntype=type_, feat=feat, ndata=value, update_links=False)
            else:
                if type_ == '_g':
                    value = value.unsqueeze()

                _data[feat] = value

    def elems_feats(self, type_=None, feats=None, values=None, **kwargs):
        """
        Access multiple features of elements.

        Args:
            type_ (str): The type of the elements.
            feats (list): The feature names.
            values (list): The values to set for the features.
            **kwargs: Additional keyword arguments.

        Returns:
            The feature values.
        """

        if isinstance(feats, dict):
            values = list(feats.values())
            type_, feats = list(zip(*list(feats.keys())))

        if feats is None:
            data = self.nodes(type_).data
            n = self.number_of_(type_)
            feats = list(data.keys())

        if values is None:
            feats = th.cat([self.elems_feat(_feat, type_=type_, **kwargs) 
                            for _feat in feats], dim=1)
            return feats
        else:
            if isinstance(type_, str):
                type_ = [type_] * len(feats)

            # break up tensor into separate feats
            if isinstance(values, th.Tensor) and len(feats) == values.shape[1]:
                values = th.split(values, [1] * len(feats), dim=1)

            [self.elems_feat(feat=_feat, type_=_type, 
                             value=_val, **kwargs) for _feat, _val, _type in zip(feats, values, type_)]

    def elems_feats_names(self, type_='v', saved_feats_only=False):
        """
        Get the names of the features of elements.

        Args:
            type_ (str): The type of the elements.
            saved_feats_only (bool): Whether to return only saved features.

        Returns:
            list: The names of the features.
        """
        feats = list(self.nodes_data[type_].keys())
        if not saved_feats_only:
            feats += list(self.registered_feats(type_))
        return list(set(feats))

    # Convenience Access
    # ==============================================================================

    def verts_feat(self, feat, value=None, **kwargs):
        """
        Access a feature of vertices.

        Args:
            feat (str): The feature name.
            value: The value to set for the feature.
            **kwargs: Additional keyword arguments.

        Returns:
            The feature value.
        """
        return self.elems_feat(feat=feat, value=value, type_='v', **kwargs)

    def verts_feats(self, feats=None, values=None, **kwargs):
        """
        Access multiple features of vertices.

        Args:
            feats (list): The feature names.
            values (list): The values to set for the features.
            **kwargs: Additional keyword arguments.

        Returns:
            The feature values.

    def edges_feat(self, feat, value=None, default=None, **kwargs):
        """
        Access a feature of edges.

            default: The default value for the feature.

            The feature value.

        return self.elems_feat(feat=feat, value=value, type_='e', default=default, **kwargs)

    def edges_feats(self, feats=None, values=None, **kwargs):
        """
        Access multiple features of edges.

        Args:
            feats (list): The feature names.
            values (list): The values to set for the features.
            **kwargs: Additional keyword arguments.

        Returns:
            The feature values.

        return self.elems_feats(feats=feats, values=values, type_='e', **kwargs)

    def true_edges_feat(self, feat, value=None, default=None, **kwargs):
        """
        Access a feature of true edges.

        Args:
            feat (str): The feature name.
            value: The value to set for the feature.
            default: The default value for the feature.
            **kwargs: Additional keyword arguments.

        Returns:
            The feature value.
        """
        e_val   = None
        e_is_e  = self.feat.e._is_edge.flatten()
        if value is not None:
            if not isinstance(value, th.Tensor):
                value = th.tensor(value).to(self.device)

            # FIXME (always wipe the existing values with zeros?)
            e_val   = th.zeros((self.number_of_('e'),) + value.shape[1:])
            e_val[e_is_e] = value

        e_val = self.elems_feat(feat=feat, value=e_val, type_='e', **kwargs)

        if value is None:
            e_val  = e_val[e_is_e]

        return e_val

    def true_edges_feats(self, feats=None, values=None, **kwargs):
        """
        Access multiple features of true edges.

        Args:
            feats (list): The feature names.
            values (list): The values to set for the features.
            **kwargs: Additional keyword arguments.

        Returns:
            The feature values.
        """
        e_vals = None
        e_is_e  = self.feat.e._is_edge.flatten()
        if values is not None:
            e_vals  = self.feat.e.zero.repeat(1, *values.shape[1:])
            e_vals[e_is_e] = values
        e_vals = self.elems_feats(feats=feats, values=e_vals, type_='e', **kwargs)
        if values is None:
            e_vals  = e_vals[e_is_e]
        return e_vals

        default: The default value for the feature.
        recompute (bool): Whether to recompute the feature.

    Returns:
        The feature value.
    """

    n = self.number_of_(type_ if not type_.startswith('_') else type_[1:])
    _data = self.nodes[type_].data

    if keys is not None:
        idx = self.keys_to_indices(type_=type_, keys=keys)
    if idx is not None:
        idx = th.tensor(idx, device=self.device, dtype=th.long) \
              if not isinstance(idx, th.Tensor) else idx

    # -----------------------------------------------------------------------------
    # Getter
    # -----------------------------------------------------------------------------

    if value is None:
        compute = recompute if recompute is not None else None

        # Try getting the feature
        value = _data.get(str(feat))

        # if features are not found:
        if value is None:
            compute = self.has_feat_func(type_, feat)
            init_feat = self._feat_can_be_init(type_, feat, default)
        # if features are found:
        else:
            compute = self._feat_needs_calc(type_, feat)
            init_feat = False

        if compute:
            func = self._REGISTERED_FEATS_FUNCS[type_][feat]
            func = getattr(self, func)
            value = func()

        if value is None and init_feat:
            value = self.feat_default(feat=feat, type_=type_, 
                                      default=default, repeat=True,
                                      raise_error=False)

        if value is None:
            raise KeyError("Cannot find '{}' feat: {}".format(type_, feat))
        else:
            if compute:
                if func.__dict__.get('autoset', True) is True:
                    if type_ == '_g' and len(value.shape) >= 2 and value.shape[0] != 1:
                        _data[feat] = value.unsqueeze(0)
                    else:
                        _data[feat] = value
                if func.__dict__.get('return_res', True) is False:
                    return _data[feat]

            else:
                if type_ == '_g' and len(value.shape) >= 2 and value.shape[0] == 1:
                    value = value.squeeze(0)
            if idx is not None:
                value = value[idx]
            return value

    # -----------------------------------------------------------------------------
    # Setter
    # -----------------------------------------------------------------------------

    else:
        if not isinstance(value, th.Tensor):
            value = th.tensor(value).to(self.device)

        n_ = idx.shape[0] if idx is not None else n
        d = len(value.shape)

        if value.numel() == 1 and n_ > 1:
            value = th.ones(n_, 1).to(self.device) * value

        if d <= 2:
            value = value.reshape(n_, -1)
        else:
            assert n_ == value.shape[0]
            pass

        # only allow instantiation if index provided? why this again?
        value_ = value
        if idx is not None:
            if feat not in _data:
                value_ = th.zeros(n, value.shape[1]).to(value.dtype, value.device)
            value_[idx] = value

        if 'g' not in type_:
            self.update_node_data(ntype=type_, feat=feat, ndata=value, update_links=False)
        else:
            if type_ == '_g':
                value = value.unsqueeze()

            _data[feat] = value

    def elems_feats(self, type_=None, feats=None, values=None, **kwargs):
    """
    Access multiple features of elements.

    Args:
        type_ (str): The type of the elements.
        feats (list): The feature names.
        values (list): The values to set for the features.
        **kwargs: Additional keyword arguments.

    Returns:
        The feature values.
    """

    if isinstance(feats, dict):
        values = list(feats.values())
        type_, feats = list(zip(*list(feats.keys())))

    if feats is None:
        data = self.nodes(type_).data
        n = self.number_of_(type_)
        feats = list(data.keys())

    if values is None:
        feats = th.cat([self.elems_feat(_feat, type_=type_, **kwargs) 
                        for _feat in feats], dim=1)
        return feats
    else:
        if isinstance(type_, str):
            type_ = [type_] * len(feats)

        # break up tensor into separate feats
        if isinstance(values, th.Tensor) and len(feats) == values.shape[1]:
            values = th.split(values, [1] * len(feats), dim=1)

        [self.elems_feat(feat=_feat, type_=_type, 
                         value=_val, **kwargs) for _feat, _val, _type in zip(feats, values, type_)]

    """
