    @classmethod
    def _initialize_CWG(cls, ltypes, n_nodes=None): 
        ltypes = [_lt for _lts in list(ltypes.values()) for _lt in _lts] 
        CWG = empty_heterograph_from_ltypes(ltypes)                                                              
        CWG.add_nodes(1, ntype='g')
        CWG.add_nodes(1, ntype='_g')
        return CWG

    def number_of_(self, type='v'):
        num_map = self._number_of()
        if type not in num_map:
            num_map.update({type: lambda: self.number_of_nodes(type)})
        return num_map[type]()

    def add_CWG_links(self, ltype, link_nodes, **ldata):
        ntype_src, ntype_lnk, ntype_dst = ltype

        nidx_unpacked = list(map(lambda x: th.tensor(x).to(self.device) \
                                           if not any([_x is None for _x in x]) else None,
                                 zip(*link_nodes)))
        nidx_src, nidx_lnk, nidx_dst = nidx_unpacked

        assert nidx_src.shape[0] == nidx_dst.shape[0]
        if nidx_lnk is not None:
            nidx_lnk.shape[0] == nidx_src.shape[0]

        n_lnks = self.number_of_links(ltype=ltype)
        # # WHY
        # if n_lnks > 0:
        #     CWG.remove_edges(th.arange(n_adjs), etype=ltype)
        self.add_links(nidx_src, nidx_dst, ltype=ltype)

        if nidx_lnk is not None:
            self.edges[ltype].data['lid'] = nidx_lnk.to(self.device)
        if ldata is not None:
            _ = [self.edges[ltype].data.__setitem__(_k, _v) for _k, _v in ldata.items()]
