    class CompasGraphInteroperability:
    """Class for interoperability with CompasGraph."""

    @classmethod
    def from_graph(cls, graph,
                    nks=None, eks=None,
                    nattrs=None, eattrs=None,
                    update_attrs=True):
        """
        Create an instance from a graph.

        Args:
            graph: The input graph.
            nks: Node keys (optional).
            eks: Edge keys (optional).
            nattrs: Node attributes (optional).
            eattrs: Edge attributes (optional).
            update_attrs: Whether to update attributes (default: True).

        Returns:
            An instance of CompasGraphInteroperability.
        """

        if nks is None:
            nks = list(graph.nodes())
        if eks is None:
            eks = list(graph.edges())
        eks = graph.edges_in_node_set(nks)

        # Collect Number-like Attributes: boolean, float, int
        # --------------------------------------------------------------------------
        # TODO: Replace with actual defaults here
        if nattrs is None:
            nattrs = graph.default_node_attributes.keys() 
        if eattrs is None:
            eattrs = graph.default_edge_attributes.keys() 

        graph_ = graph.to_subgraph(nks, eks, nattrs, eattrs)

        nks = graph_.nks
        eks = graph_.eks

        n_n = len(nks)
        n_e = len(eks)

        ndat = [list(_l) for _l in list(zip(*graph_.nodes_attributes(keys=nks, names=nattrs)))]
        edat = [list(_l) for _l in list(zip(*graph_.edges_attributes(keys=eks, names=eattrs)))]

        CWG = cls()
        CWG.add_nodes(n_n, ntype='v')
        CWG.add_nodes(n_e, ntype='e')

        for _i_n_a, _n_a in enumerate(nattrs):
            CWG.feat.v[_n_a] = _cast_data(ndat[_i_n_a])
        for _i_e_a, _e_a in enumerate(eattrs):
            CWG.feat.e[_e_a] = _cast_data(edat[_i_e_a])

        CWG.feat.v._key = th.arange(n_n).view(-1,1)
        CWG.feat.e._key = th.tensor(eks)
        CWG.setup_local_adjacencies(graph_)
        return CWG

    def setup_local_adjacencies(self, G):
        """
        Setup local adjacencies for the graph.

        Args:
            G: The input graph.
        """
        self._init_0CWC_X(G)
        self._init_0CWC_C_e_v(G)
        self._init_0CWC_U_vev(G)
        self._init_1CWC_B_v_e(G)
        self._init_1CWC_L_eve(G)

    # ================================================================================        
    # 0-CWComplex
    # ================================================================================

    def _init_0CWC_X(self, G):
        """
        Initialize 0-CWComplex X.

        Args:
            G: The input graph.
        """
        n_v = self.number_of_('v')
        n_e = self.number_of_('e')
        lnis = list(zip(list(range(n_v)), [None] * n_v, [0] * n_v))
        self.add_CWG_links(('v', '', 'g'), link_nodes=lnis)  
        lnis = list(zip(list(range(n_e)), [None] * n_e, [0] * n_e))
        self.add_CWG_links(('e', '', 'g'), link_nodes=lnis)

    def _init_0CWC_C_e_v(self, G):
        """
        Initialize 0-CWComplex C_e_v.

        Args:
            G: The input graph.
        """
        lnk_nis_e_v = []

        vis = G.nks
        vi_to_nbrs = {_vi: G.node_neighbors(_vi) for _vi in vis} 
        evi_to_ei = G.ek_to_ei_r

        for _vi_i in vis:
            for _vi_j in vi_to_nbrs[_vi_i]:
                _evi = (_vi_j, _vi_i)
                _ei = evi_to_ei[_evi]
                lnk_nis_e_v.append([_ei, None, _vi_i])

        self.add_CWG_links(('e', '', 'v'), link_nodes=lnk_nis_e_v)  

    def _init_0CWC_U_vev(self, G):
        """
        Initialize 0-CWComplex U_vev.

        Args:
            G: The input graph.
        """
        lidx_vev = []

        vis = G.nks
        evi_to_ei = G.ek_to_ei_r
        vi_to_vis = {_vi: G.node_neighbors(_vi) for _vi in vis} 

        for _vi_i in vis:
            for _vi_j in vi_to_vis[_vi_i]:
                _evi = (_vi_j, _vi_i)
                _ei = evi_to_ei[_evi]
                # link vertex cell to edge cell to vertex cell
                lidx_vev.append([_vi_j, _ei, _vi_i])

        self.add_CWG_links(('v', 'e', 'v'), link_nodes=lidx_vev)

    # ================================================================================        
    # 1-CWComplex
    # ================================================================================

    def _init_1CWC_B_v_e(self, G):
        """
        Initialize 1-CWComplex B_v_e.

        Args:
            G: The input graph.
        """
        lnk_nis_v_e = []

        evi_to_ei = G.ek_to_ei_r
        evis = G.eks

        for _evi in evis:  
            # link vertex cell to edge cell
            _ei = evi_to_ei[_evi]
            lnk_nis_v_e.append(([_evi[0], None, _ei]))
            lnk_nis_v_e.append(([_evi[1], None, _ei]))

        self.add_CWG_links(('v', '', 'e'), link_nodes=lnk_nis_v_e)

    def _init_1CWC_L_eve(self, G):
        """
        Initialize 1-CWComplex L_eve.

        Args:
            G: The input graph.
        """
        lnk_nis_eve = []

        evi_to_ei = G.ek_to_ei_r
        evis = G.eks
        evi_to_vi_to_evis_nbr_low = {_evi: G.edge_neighbors(_evi) for _evi in evis}

        for _evi_i in evis: # source edge
            _ei_i = evi_to_ei[_evi_i]
            _vi_to_evis_nbr = evi_to_vi_to_evis_nbr_low[_evi_i]
            for _vi, _evi_js in _vi_to_evis_nbr.items():  # edge to lower neighbours
                for _evi_j in _evi_js:
                    _ei_j = evi_to_ei[_evi_j]
                    lnk_nis_eve.append([_ei_i, _vi, _ei_j])

        self.add_CWG_links(('e', 'v', 'e'), link_nodes=lnk_nis_eve)

    class CompasHalfEdgeInteroperability:
    """Class for interoperability with CompasHalfEdge."""

    @classmethod
    def from_mesh(cls, mesh,
                  vks=None, eks=None,  fks=None,
                  vattrs=None, eattrs=None, fattrs=None,
                  update_attrs=True, save_mesh_data=False, 
                  is_formforce=False):
        """
        Create an instance from a mesh.

        Args:
            mesh: The input mesh.
            vks: Vertex keys (optional).
            eks: Edge keys (optional).
            fks: Face keys (optional).
            vattrs: Vertex attributes (optional).
            eattrs: Edge attributes (optional).
            fattrs: Face attributes (optional).
            update_attrs: Whether to update attributes (default: True).
            save_mesh_data: Whether to save mesh data (default: False).
            is_formforce: Whether it is a formforce mesh (default: False).

        Returns:
            An instance of CompasHalfEdgeInteroperability.
        """

        # convert to simplified form diagram embedded in geom_ml
        # opt: reassign keys based on elements included
        # get relevant feats
        # set halfedge info
        # proceed with individual adjacency setup

        # Elements Keys-to-Indices Mapping
        # --------------------------------------------------------------------------

        if fks is None:
            fks = list(mesh.faces())

        if eks is None:
            eks = list(mesh.edges())

        if vks is None:
            vks = sorted(list(set([_vk for _ek in eks for _vk in _ek])))

        # Collect Number-like Attributes: boolean, float, int
        # --------------------------------------------------------------------------
        # TODO: Replace with actual defaults here
        if vattrs is None:
            vattrs = ['is_fixed', 'x', 'y', 'z']
        if eattrs is None:
            eattrs = ['_is_edge', 'q']
        if fattrs is None:
            fattrs = ['_is_loaded']

        # Builds a clean sub-mesh with reordered vertices
        # --------------------------------------------------------------------------
        mesh_ = mesh.to_submesh(vks, eks, fks, vattrs, eattrs, fattrs)

        vks = mesh_.vks
        eks = mesh_.eks
        fks = mesh_.fks

        vdat = [list(_l) for _l in list(zip(*mesh_.vertices_attributes(keys=vks, names=vattrs)))]
        edat = [list(_l) for _l in list(zip(*mesh_.edges_attributes(keys=eks, names=eattrs)))]
        fdat = [list(_l) for _l in list(zip(*mesh_.faces_attributes(keys=fks, names=fattrs)))]

        n_v = len(vks)
        n_e = len(eks)
        n_f = len(fks)

        # --------------------------------------------------------------------------
        # --------------------------------------------------------------------------
        # Create CWG instance
        # --------------------------------------------------------------------------

        CWG = cls()

        # --------------------------------------------------------------------------

        CWG.add_nodes(n_v, ntype='v')
        CWG.add_nodes(n_e, ntype='e')
        CWG.add_nodes(n_f, ntype='f')

        for _i_v_a, _v_a in enumerate(vattrs):
            CWG.feat.v[_v_a] = th.tensor(vdat[_i_v_a])
        for _i_e_a, _e_a in enumerate(eattrs):
            CWG.feat.e[_e_a] = th.tensor(edat[_i_e_a])
        for _i_f_a, _f_a in enumerate(fattrs):
            CWG.feat.f[_f_a] = th.tensor(fdat[_i_f_a])

        CWG.feat.v._key = th.tensor(vks).to(th.long)
        CWG.feat.e._key = th.tensor(eks).to(th.long)
        CWG.feat.f._key = th.tensor(fks).to(th.long)

        CWG.setup_local_adjacencies(mesh_)

        CWG.edge        = list(mesh_.edges())
        CWG.face        = mesh_.face
        CWG.halfedge    = mesh_.halfedge
        return CWG

    def setup_local_adjacencies(self, G):
        """
        Setup local adjacencies for the mesh.

        Args:
            G: The input mesh.
        """
        # since mesh is cleaned up, keys and indices are interchangable...
        self._init_0CWC_C_e_v(G)
        self._init_0CWC_U_vev(G)
        self._init_1CWC_B_v_e(G)
        self._init_1CWC_C_f_e(G)

    def _init_0CWC_C_e_v(self, G):
        """
        Initialize 0-CWComplex C_e_v.

        Args:
            G: The input mesh.
        """
        lnk_nis_e_v = []

        vis = G.vks
        vi_to_nbrs = {_vi: G.vertex_neighbors(_vi) for _vi in vis} 
        evi_to_ei = G.ek_to_ei_r

        for _vi_i in vis:
            for _vi_j in vi_to_nbrs[_vi_i]:
                _evi = (_vi_j, _vi_i)
                _ei = evi_to_ei[_evi]
                lnk_nis_e_v.append([_ei, None, _vi_i])

        self.add_CWG_links(('e', '', 'v'), link_nodes=lnk_nis_e_v)  


    def _init_0CWC_U_vev(self, G):
        """
        Initialize 0-CWComplex U_vev.

        Args:
            G: The input mesh.
        """

        lidx_vev = []

        vis = G.vks
        evi_to_ei = G.ek_to_ei_r
        vi_to_vis = {_vi: G.vertex_neighbors(_vi) for _vi in vis}  #MESH-ONLY

        hevi_fi = []
        for _vi_i in vis:
            for _vi_j in vi_to_vis[_vi_i]:
                _evi = (_vi_j, _vi_i)
                _ei = evi_to_ei[_evi]
                # link vertex cell to edge cell to vertex cell
                lidx_vev.append([_vi_j, _ei, _vi_i])
                # include corresponding halfedge information  # MESH-ONLY
                _fi = G.halfedge[_vi_j][_vi_i]
                hevi_fi.append(_fi if _fi is not None else -1)

        hevi_fi = th.tensor(hevi_fi).to(th.long)
        self.add_CWG_links(('v', 'e', 'v'), link_nodes=lidx_vev, hefi=hevi_fi)

        evi_to_vi_to_evis_nbr_low = {_evi: G.edge_lower_neighbors(_evi) for _evi in evis}

    def _init_2CWC_X(self, G):
        """
        Initialize 2-CWComplex X.

        Args:
            G: The input mesh.
        """

        vi_to_fis_nbr = G.vk_to_fks

        lnk_nis_v_f = []
        lnk_nis_f_v = []
        for _vi, _fis_nbr in vi_to_fis_nbr.items():
            for _fi in _fis_nbr:
                lnk_nis_v_f.append([_vi, None, _fi])
                lnk_nis_f_v.append([_fi, None, _vi])

        self.add_CWG_links(('v', '', 'f'), lnk_nis_v_f)
        self.add_CWG_links(('f', '', 'v'), lnk_nis_f_v)

        n_v = self.number_of_('v')
        n_e = self.number_of_('e')
        n_e = self.number_of_('f')
        lnis = list(zip(list(range(n_v)), [None] * n_v, [0] * n_v))
        self.add_CWG_links(('v', '', 'g'), link_nodes=lnis)  
        lnis = list(zip(list(range(n_e)), [None] * n_e, [0] * n_e))
        self.add_CWG_links(('e', '', 'g'), link_nodes=lnis)
        lnis = list(zip(list(range(n_e)), [None] * n_e, [0] * n_e))
        self.add_CWG_links(('f', '', 'g'), link_nodes=lnis)

        if export_feats:
            vk_to_attr_dict = self.key_to_attr_dict('v')
            ek_to_attr_dict = self.key_to_attr_dict('e')
            fk_to_attr_dict = self.key_to_attr_dict('f')
        else:
            vks = self.keys('v')

            ek_to_attr_dict = {_ek: {} for _ek in eks}
            fk_to_attr_dict = {_fk: {} for _fk in fks}

        fi_to_vis = self.face 

        fi_to_fk = self.fi_to_fk
        vi_to_vk = self.vi_to_vk
        fk_to_vks = {fi_to_fk[_fi]: [vi_to_vk[_vi] for _vi in _vis] for _fi, _vis in fi_to_vis.items()}

        [mesh.add_vertex(key=_vk, attr_dict=_attr_dict)
         for _vk, _attr_dict in vk_to_attr_dict.items()]
        [mesh.add_face(vertices=fk_to_vks[_fk], fkey=_fk, attr_dict=_attr_dict)

        [mesh.edgedata.update({str(tuple(sorted(_ek))): _attr_dict})
         for _ei, (_ek, _attr_dict) in enumerate(ek_to_attr_dict.items())]

        return mesh
