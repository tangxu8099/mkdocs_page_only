    class CWGConstrainedForceDensitySupport(CWGForceDensitySupport):

    @property
    def n_e_ind(self):
        """
        Returns the number of independent edges.

        Returns:
            int: Number of independent edges.
        """
        return self.feat.e._is_ind.sum().item()

    @property
    def ee_is_ind(self):
        """
        Returns a tensor indicating which edges are independent.

        Returns:
            torch.Tensor: Tensor indicating independent edges.
        """
        assert self.feat.g._has_ind
        return self.feat.e._is_ind[self.feat.e._is_edge.flatten()]

    @property
    def ee_is_dep(self):
        """
        Returns a tensor indicating which edges are dependent.

        Returns:
            torch.Tensor: Tensor indicating dependent edges.
        """
        return ~self.ee_is_ind

    @property
    def ee_q_ind(self):
        """
        Returns the force densities of the independent edges.

        Returns:
            torch.Tensor: Force densities of independent edges.
        """
        return self.feat.e.q[self.eeis_ind]

    @property
    def ee_q_dep(self):
        """
        Returns the force densities of the dependent edges.

        Returns:
            torch.Tensor: Force densities of dependent edges.
        """
        return self.feat.e.q[self.eeis_dep]

    @property
    def eis_dep(self):
        """
        Returns the indices of the dependent edges.

        Returns:
            torch.Tensor: Indices of dependent edges.
        """
        assert self.feat.g._has_ind
        n_e = self.n_e
        e_msk = th.logical_and(self.feat.e._is_edge, ~self.feat.e._is_ind).flatten()
        return th.arange(n_e)[e_msk]

    @property
    def eis_ind(self):
        """
        Returns the indices of the independent edges.

        Returns:
            torch.Tensor: Indices of independent edges.
        """
        assert self.feat.g._has_ind
        return self.eis[self.feat.e._is_ind.flatten()]

    @property
    def eeis_ind(self):
        """
        Returns the indices of the independent edges in the edge-edge adjacency matrix.

        Returns:
            torch.Tensor: Indices of independent edges in the edge-edge adjacency matrix.
        """
        n_ee = self.n_ee
        return th.arange(n_ee, device=self.device)[self.ee_is_ind.flatten()]

    @property
    def eeis_dep(self):
        """
        Returns the indices of the dependent edges in the edge-edge adjacency matrix.

        Returns:
            torch.Tensor: Indices of dependent edges in the edge-edge adjacency matrix.
        """
        assert self.feat.g._has_ind
        n_ee = self.n_ee
        return th.arange(n_ee, device=self.device)[self.ee_is_dep.flatten()]

    @property
    def eei_to_ei(self):
        """
        Returns a dictionary mapping edge-edge indices to edge indices.

        Returns:
            dict: Dictionary mapping edge-edge indices to edge indices.
        """
        return dict(zip(list(range(self.n_ee)), self.eis[self.feat.e._is_edge.flatten()].tolist()))

    @register_feat_func(feat='E', elem_type='_g', autoset=True, is_property=True)
    def _E(self):
        """
        Computes the equilibrium matrix.

        Returns:
            torch.Tensor: Equilibrium matrix.
        """
        Citx = self.Ct[self.vis_freed.flatten(), :]  
        City = self.Ct[self.vis_freed.flatten(), :]
        uvw = self.uvw
        U = th.diag(uvw[:, 0].flatten()).to(self.device)
        V = th.diag(uvw[:, 1].flatten()).to(self.device)
        E = th.vstack((th.mm(Citx, U), th.mm(City, V))).to(self.device)
        return E

    @register_feat_func(feat='E_rank', elem_type='g', autoset=True, is_property=True)
    def _E_rank(self):
        """
        Computes the rank of the equilibrium matrix.

        Returns:
            torch.Tensor: Rank of the equilibrium matrix.
        """
        E = self.E
        return th.linalg.matrix_rank(E).reshape(1, 1)

    @property
    def has_independent_edges(self):
        """
        Checks if the graph has independent edges.

        Returns:
            bool: True if the graph has independent edges, False otherwise.
        """
        if self.has_feat(type_='e', feat='_is_ind') and \
           self.has_feat(type_='_g', feat='E'):
            return True
        return False

    def number_of_independent_edges(self):
        """
        Returns the number of independent edges.

        Returns:
            int: Number of independent edges.
        """
        if self.has_independent_edges:
            return self.number_of_true_edges() - (~self.feat.ee._is_ind).sum()
        else:
            E = self.E
            return E.shape[-1] - th.linalg.matrix_rank(E)

    def save_independent_edges(self, eis_ind=None, eis_edge_ind=None):
        """
        Saves the independent edges.

        Args:
            eis_ind (list, optional): List of independent edge indices. Defaults to None.
            eis_edge_ind (list, optional): List of edge-edge independent edge indices. Defaults to None.

        Returns:
            torch.Tensor: Tensor indicating independent edges.
        """
        e_is_ind = self.feat.e.zero.to(th.bool)
        if eis_edge_ind is not None:
            eis_ind = self.eis_edge[eis_edge_ind]
        e_is_ind[eis_ind] = True

        self.feat.e._is_ind = e_is_ind
        self.feat.g._has_ind = True

        for _k in list(self.nodes['_g'].data.keys()):
            if _k in ['Ed', 'Ed_inv', 'Ei']:
                _ = self.nodes['_g'].data.pop(_k)
        return e_is_ind

    def process_edge_input(self, eks_in, eis_in, eeis_in, out='eei'):
        """
        Processes the edge input.

        Args:
            eks_in (list): List of edge keys.
            eis_in (list): List of edge indices.
            eeis_in (list): List of edge-edge indices.
            out (str, optional): Output type. Defaults to 'eei'.

        Returns:
            list: Processed edge-edge indices.
        """
        if eks_in:
            eeis_in = self.keys_to_indices(type_='e', keys=eks_in)
        if eis_in is not None:
            eeis_in = self.eis_to_eeis(eis_in)
        return eeis_in

    def edges_are_independent(self, eks=None, eis=None, eeis=None):
        """
        Checks if the edges are independent.

        Args:
            eks (list, optional): List of edge keys. Defaults to None.
            eis (list, optional): List of edge indices. Defaults to None.
            eeis (list, optional): List of edge-edge indices. Defaults to None.

        Returns:
            bool: True if the edges are independent, False otherwise.
        """
        E = self.E
        self.number_of_independent_edges()

        eeis_ind = eis
        eeis_ind = self.process_edge_input(eks_in=eks, eis_in=eis, eeis_in=eeis)

        rnk = th.linalg.matrix_rank(E)
        _, m = E.shape
        n_ind = m - rnk

        E_dep_rnk_1 = self.E_slice_rank(exclude=True, eeis=eeis_ind)
        eeis_dep = self.complementary_indices('ee', eeis_ind)
        _, _, E_dep_rnk_2 = self._iteratively_build_E(E, eeis=eeis_dep)

        print('E rank: {} | E_dep_test: rank 1: {}, rank 2: {}'.format(rnk, E_dep_rnk_1, E_dep_rnk_2))
        if rnk == E_dep_rnk_1 and rnk == E_dep_rnk_2:
            return True
        return False

    def _iteratively_build_E(self, E, eeis):
        """
        Iteratively builds the equilibrium matrix.

        Args:
            E (torch.Tensor): Equilibrium matrix.
            eeis (list): List of edge-edge indices.

        Returns:
            tuple: Independent edge-edge indices, dependent equilibrium matrix, and rank.
        """
        E = E if E is not None else self.E
        _, m = E.shape
        E_temp = E[:, []]

        eeis_ind = []

        for eei in eeis:
            E_test = th.hstack([E_temp, E[:, [eei]]])
            _, ncol = E_test.shape
            r = th.linalg.matrix_rank(E_test)
            #print("ei: {} | r: {} | n_col: {}".format(eei, r, ncol))

            if r < ncol:
                #print('added: {}'.format(eei))
                eeis_ind.append(eei)
            else:
                E_temp = E_test

        E_dep = E_temp

        return eeis_ind, E_dep, r

    def find_independent_edges(self, E=None, eis=None, eeis=None,
                               eis_ind=None, eeis_ind=None,
                               randomize=False, save=True):
        """
        Finds the independent edges.

        Args:
            E (torch.Tensor, optional): Equilibrium matrix. Defaults to None.
            eis (list, optional): List of edge indices. Defaults to None.
            eeis (list, optional): List of edge-edge indices. Defaults to None.
            eis_ind (list, optional): List of independent edge indices. Defaults to None.
            eeis_ind (list, optional): List of independent edge-edge indices. Defaults to None.
            randomize (bool, optional): Whether to randomize the edges. Defaults to False.
            save (bool, optional): Whether to save the independent edges. Defaults to True.

        Returns:
            list: Independent edge-edge indices.
        """
        from random import shuffle

        if eis is None and eeis is None:
            eeis = self.indices(type_='ee')
        else:
            eeis = self.process_edge_input(eks_in=None, eis_in=eis, eeis_in=eeis)

        if eis_ind is not None or eeis_ind is not None:
            eeis_ind = self.process_edge_input(eks_in=None, eis_in=eis_ind, eeis_in=eeis_ind)
            eeis = list(set(eeis) - set(eeis_ind)) + eeis_ind

        if randomize:
            shuffle(eeis)

        eeis_ind, _, _ = self._iteratively_build_E(E, eeis)
        eeis_ind = sorted(eeis_ind)

        if save:
            self.save_independent_edges(eis_edge_ind=eeis_ind)

        return eeis_ind

    def is_in_equilibrium(self, mode='both'):
        """
        Checks if the graph is in equilibrium.

        Args:
            mode (str, optional): Mode of equilibrium ('both', 'h', 'v'). Defaults to 'both'.

        Returns:
            bool: True if the graph is in equilibrium, False otherwise.
        """
        eqs = []
        eqs += ([self.is_in_heq] if 'h' in mode else []) 
        eqs += ([self.is_in_veq] if 'v' in mode else [])
        return all(eqs)

    def solve_independent_edges(self):
        """
        Solves for the independent edges.
        """
        self.find_independent_edges(save=True)
        self.Ed_inv

    def solve_dependent_q(self, e_q_ind=None, save=True):
        """
        Solves for the dependent force densities.

        Args:
            e_q_ind (torch.Tensor, optional): Independent force densities. Defaults to None.
            save (bool, optional): Whether to save the force densities. Defaults to True.

        Returns:
            torch.Tensor: Dependent force densities.
        """
        e_q = self.feat.e.zero
        if e_q_ind is not None:
            e_q[self.eis_ind] = e_q_ind
        else:
            e_q[self.eis_ind] = self.feat.e.q[self.eis_ind]

        e_q_ind = e_q[self.eis_ind]
        e_q_dep = -th.matmul(self.Ed_inv, th.matmul(self.Ei, e_q_ind))
        e_q[self.eis_dep] = e_q_dep

        if save:
            self.feat.e.q = e_q

        return e_q_dep

    @register_feat_func(feat='Ei', elem_type='_g', 
                        autoset=True, is_property=True)
    def _Ei(self):
        """
        Computes the equilibrium matrix for the independent edges.

        Returns:
            torch.Tensor: Equilibrium matrix for the independent edges.
        """
        assert self.feat.g._has_ind
        return self.E[:, self.eeis_ind]

    @register_feat_func(feat='Ed', elem_type='_g', 
                        autoset=True, is_property=True)
    def _Ed(self):
        """
        Computes the equilibrium matrix for the dependent edges.

        Returns:
            torch.Tensor: Equilibrium matrix for the dependent edges.
        """
        assert self.feat.g._has_ind
        return self.E[:, self.eeis_dep]

    @register_feat_func(feat='Ed_inv', elem_type='_g', 
                        autoset=True, is_property=True)
    def _Ed_inv(self):
        """
        Computes the pseudo-inverse of the equilibrium matrix for the dependent edges.

        Returns:
            torch.Tensor: Pseudo-inverse of the equilibrium matrix for the dependent edges.
        """
        assert self.feat.g._has_ind
        return th.linalg.pinv(self.Ed)

    def E_slice(self, eis=[], eeis=[], exclude=False):
        """
        Slices the equilibrium matrix.

        Args:
            eis (list, optional): List of edge indices. Defaults to [].
            eeis (list, optional): List of edge-edge indices. Defaults to [].
            exclude (bool, optional): Whether to exclude the specified indices. Defaults to False.

        Returns:
            torch.Tensor: Sliced equilibrium matrix.
        """
        if eis:
            eeis = self.eis_to_eeis(eis)
        if exclude:
            eeis = self.complementary_indices('ee', eeis)

        return self.E[:, eeis]

    def E_slice_rank(self, **kwargs):
        """
        Computes the rank of the sliced equilibrium matrix.

        Args:
            **kwargs: Additional keyword arguments.

        Returns:
            torch.Tensor: Rank of the sliced equilibrium matrix.
        """
        E_slc = self.E_slice(**kwargs)
        return th.linalg.matrix_rank(E_slc)

    def E_slice_cond(self, **kwargs):
        """
        Computes the condition number of the sliced equilibrium matrix.

        Args:
            **kwargs: Additional keyword arguments.

        Returns:
            torch.Tensor: Condition number of the sliced equilibrium matrix.
        """
        E_slc = self.E_slice(**kwargs)
        return th.linalg.cond(E_slc)

    def find_independent_edges(self, E=None, eis=None, eeis=None,
                           eis_ind=None, eeis_ind=None,
                           randomize=False, save=True):
    """
    Finds the independent edges.

    Args:
        E (torch.Tensor, optional): Equilibrium matrix. Defaults to None.
        eis (list, optional): List of edge indices. Defaults to None.
        eeis (list, optional): List of edge-edge indices. Defaults to None.
        eis_ind (list, optional): List of independent edge indices. Defaults to None.
        eeis_ind (list, optional): List of independent edge-edge indices. Defaults to None.
        randomize (bool, optional): Whether to randomize the edges. Defaults to False.
        save (bool, optional): Whether to save the independent edges. Defaults to True.

    Returns:
        list: Independent edge-edge indices.
    """
    from random import shuffle

    if eis is None and eeis is None:
        eeis = self.indices(type_='ee')
    else:
        eeis = self.process_edge_input(eks_in=None, eis_in=eis, eeis_in=eeis)

    if eis_ind is not None or eeis_ind is not None:
        eeis_ind = self.process_edge_input(eks_in=None, eis_in=eis_ind, eeis_in=eeis_ind)
        eeis = list(set(eeis) - set(eeis_ind)) + eeis_ind

    if randomize:
        shuffle(eeis)

    eeis_ind, _, _ = self._iteratively_build_E(E, eeis)
    eeis_ind = sorted(eeis_ind)

    if save:
        self.save_independent_edges(eis_edge_ind=eeis_ind)

    return eeis_ind

    if save:
        self.feat.e.q = e_q

    return e_q_dep
