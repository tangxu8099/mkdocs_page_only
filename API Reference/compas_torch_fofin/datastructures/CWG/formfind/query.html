    class CWGCommonFormFindingQuery:
    """
    Class for common form-finding queries on a CWGraph.
    """

    def number_of_true_edges(self):
        """
        Returns the number of true edges.

        Returns:
            torch.Tensor: The number of true edges.
        """
        # HACKY
        # if '_is_edge' not in self.nodes['e'].data:
            # self.nodes['e'].data['_is_edge'] = th.ones(self.number_of_nodes('e'), 1).to(self.device)
            # return self.number_of_nodes('e')
        return self.feat.e._is_edge.sum()

    # TRUE EDGE vs EDGE
    # print('hello')
    @property
    def n_ee(self):
        """
        Returns the number of true edges.

        Returns:
            int: The number of true edges.
        """
        return self.feat.e._is_edge.flatten().sum().item()

    # ALL SPATIAL GRAPH
    @property
    def vis_fixed(self):
        """
        Returns the indices of fixed vertices.

        Returns:
            torch.Tensor: The indices of fixed vertices.
        """
        return self.vis[self.feat.v.is_fixed.flatten().to(th.bool)].to(self.device)

    @property
    def vis_freed(self):
        """
        Returns the indices of freed vertices.

        Returns:
            torch.Tensor: The indices of freed vertices.
        """
        return self.vis[~self.feat.v.is_fixed.flatten().to(th.bool)].to(self.device)

    @property
    def vis_loaded(self):
        """
        Returns the indices of loaded vertices.

        Returns:
            list: The indices of loaded vertices.
        """
        return self.vis[self.feat.v[('px', 'py', 'pz')].abs().sum(dim=1).flatten() > 0.].tolist()

    # HALFEDGE
    @property
    def vis_on_bound(self):
        """
        Returns the indices of vertices on the boundary.

        Returns:
            torch.Tensor: The indices of vertices on the boundary.
        """
        return self.vis[self.feat.v._is_on_bound.flatten().to(th.bool).to(self.device)]

    @property
    def vis_int(self):
        """
        Returns the indices of internal vertices.

        Returns:
            torch.Tensor: The indices of internal vertices.
        """
        return self.vis[~self.feat.v._is_on_bound.flatten().to(th.bool).to(self.device)]

    @property
    def vis_freed_on_bound(self):
        """
        Returns the indices of freed vertices on the boundary.

        Returns:
            torch.Tensor: The indices of freed vertices on the boundary.
        """
        return self.vis[th.logical_and(self.feat.v._is_on_bound.flatten().to(th.bool).to(self.device),
                                       ~self.feat.v.is_fixed.flatten().to(th.bool).to(self.device))]

    @property
    def fis_loaded(self):
        """
        Returns the indices of loaded faces.

        Returns:
            torch.Tensor: The indices of loaded faces.
        """
        return th.arange(self.number_of_('f'))[self.feat.f._is_loaded_.flatten()].to(self.device)

    @property
    def eeis(self):
        """
        Returns the indices of true edges.

        Returns:
            torch.Tensor: The indices of true edges.
        """
        return th.arange(self.number_of_true_edges()).to(self.device)

    @property
    def eevis(self):
        """
        Returns the edge-vertex indices of true edges.

        Returns:
            torch.Tensor: The edge-vertex indices of true edges.
        """
        return self.evis[self.feat.e._is_edge.flatten()]

    @property
    def eis_edge(self):
        """
        Returns the edge indices of true edges.

        Returns:
            torch.Tensor: The edge indices of true edges.
        """
        return self.eis[self.feat.e._is_edge.flatten()].to(self.device)

    @property
    def evis_edge(self):
        """
        Returns the edge-vertex indices of true edges.

        Returns:
            torch.Tensor: The edge-vertex indices of true edges.
        """
        return self.evis[self.feat.e._is_edge.flatten()]
        # return list(map(lambda x: tuple(x), self.evis[self.feat.e._is_edge.flatten()].tolist()))

    @property
    def ee_q(self):
        """
        Returns the force densities of true edges.

        Returns:
            torch.Tensor: The force densities of true edges.
        """
        return self.feat.e.q[self.eis_edge].to(self.device)

    @property
    def ei_to_eei(self):
        """
        Returns a dictionary mapping edge indices to true edge indices.

        Returns:
            dict: Dictionary mapping edge indices to true edge indices.
        """
        return {_v: _k for _k, _v in self.eei_to_ei.items()}

    def eeis_to_eis(self, eeis):
        """
        Converts true edge indices to edge indices.

        Args:
            eeis (list): List of true edge indices.

        Returns:
            list: List of edge indices.
        """
        eei_to_ei = self.eei_to_ei
        return [eei_to_ei[_eei] for _eei in eeis]

    def eis_to_eeis(self, eis):
        """
        Converts edge indices to true edge indices.

        Args:
            eis (list): List of edge indices.

        Returns:
            list: List of true edge indices.
        """
        ei_to_eei = self.ei_to_eei
        return [ei_to_eei[_ei] for _ei in eis]

    # =============================================================================
    # Connectivity Matrix
    # =============================================================================

    @register_feat_func(feat='C', elem_type='_g', autoset=True, is_property=True)
    def _C(self):
        """
        Computes the branch-node connectivity matrix.

        Returns:
            torch.Tensor: The branch-node connectivity matrix.
        """
        from compas_torch_fofin.matrices import branch_node_connectivity
        C = branch_node_connectivity(self.evis_edge, self.n_v)
        return C

    @register_feat_func(feat='Ci', elem_type='_g', autoset=True, is_property=True)
    def _Ci(self):
        """
        Computes the branch-node connectivity matrix for freed vertices.

        Returns:
            torch.Tensor: The branch-node connectivity matrix for freed vertices.
        """
        return self.C[:, self.vis_freed.flatten()]

    @register_feat_func(feat='Cf', elem_type='_g', autoset=True, is_property=True)
    def _Cf(self):
        """
        Computes the branch-node connectivity matrix for fixed vertices.

        Returns:
            torch.Tensor: The branch-node connectivity matrix for fixed vertices.
        """
        return self.C[:, self.vis_fixed.flatten()]

    @register_feat_func(feat='Ct', elem_type='_g', autoset=True, is_property=True)
    def _Ct(self):
        """
        Computes the transpose of the branch-node connectivity matrix.

        Returns:
            torch.Tensor: The transpose of the branch-node connectivity matrix.
        """
        return self.C.T

    @register_feat_func(feat='Cit', elem_type='_g', autoset=True, is_property=True)
    def _Cit(self):
        """
        Computes the transpose of the branch-node connectivity matrix for freed vertices.

        Returns:
            torch.Tensor: The transpose of the branch-node connectivity matrix for freed vertices.
        """
        return self.Ci.T

    @register_feat_func(feat='uvw', elem_type='_g', autoset=True, is_property=True)
    def _uvw(self):
        """
        Computes the displacements of the vertices.

        Returns:
            torch.Tensor: The displacements of the vertices.
        """
        return th.mm(self.C, self.feat.v['x', 'y', 'z'])

    check_horizontal_equilibrium = check_horizontal_equilibrium
    check_vertical_equilibrium = check_vertical_equilibrium
