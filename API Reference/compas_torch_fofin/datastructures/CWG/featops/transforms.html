    class CWGDesignTransformations(object):
    """Class for CWG Design Transformations."""

    def _transform_points_feats(self, transformation, points_feats):
        """Transform points features using a given transformation matrix.

        Args:
            transformation (torch.Tensor): The transformation matrix.
            points_feats (dict): The points features to be transformed.
        """
        for _elem_typ, _feats_gp in points_feats.items():
            for _feats in _feats_gp:
                if self.has_feats(type_=_elem_typ, feats=_feats):
                    _feats_ = self.elems_feats(type_=_elem_typ, feats=_feats)
                    _feats_ = transform_points(transformation, _feats_)
                    self.elems_feats(feats=_feats, values=_feats_, type_=_elem_typ)

    def _scale_feats(self, factor, feats):
        """Scale features by a given factor.

        Args:
            factor (float): The scaling factor.
            feats (dict): The features to be scaled.
        """
        for _elem_typ, _feats_gp in feats.items():
            for _feats in _feats_gp:
                if self.has_feats(_feats, type_=_elem_typ):
                    _feats_ = self.elems_feats(feats=_feats, type_=_elem_typ)
                    _feats_ = scale_feats(factor, _feats_)
                    self.elems_feats(feats=_feats, values=_feats_, type_=_elem_typ)

    def translate_xyz(self, translation=[0., 0., 0.], degree=True):
        """Translate points in the XY plane.

        Args:
            translation (list, optional): The translation vector. Defaults to [0., 0., 0.].
            degree (bool, optional): Whether the translation is in degrees. Defaults to True.
        """
        feats_to_tfr = {'v': [['x', 'y', 'z'],
                              ],
                        'e': [
                               ['_ctx', '_cty', '_ctz'],
                              ],
                        'f': [
                              ['_ctx', '_cty', '_ctz'],
                              ['_cnx', '_cny', '_cnz'],
                              ],
                        }
        T = th.tensor(matrix_from_translation(translation)).to(self.device)
        self._transform_points_feats(T, feats_to_tfr)

    def rotate_xy(self, angle, point=[0., 0., 0.], degree=True):
        """Rotate points in the XY plane.

        Args:
            angle (float): The rotation angle.
            point (list, optional): The point around which to rotate. Defaults to [0., 0., 0.].
            degree (bool, optional): Whether the angle is in degrees. Defaults to True.
        """
        angle = radians(angle) if degree else angle
        feats_to_tfr = {'v': [['x', 'y', 'z'],
                              ['_nx', '_ny', '_nz'],
                              ],
                        'e': [
                              ['_ctx', '_cty', '_ctz'],
                              ],
                        'f': [['_nx', '_ny', '_nz'],
                              ['_ctx', '_cty', '_ctz'],
                              ['_cnx', '_cny', '_cnz'],
                              ],
                        }
        T = th.tensor(matrix_from_axis_and_angle([0., 0., 1.], angle, point=point)).to(self.device)
        self._transform_points_feats(T, feats_to_tfr)

    def scale_xyz(self, factor=1.):
        """Scale points in the XYZ plane.

        Args:
            factor (float, optional): The scaling factor. Defaults to 1.
        """
        # pts feats rescaled by factor
        feats_to_tfr = {'v': [['x', 'y', 'z'],
                              ],
                        }
        T = th.tensor(matrix_from_scale_factors([factor, factor, factor])).to(self.device)
        self._transform_points_feats(T, feats_to_tfr)

        # feats rescaled by factor
        feats_to_tfr = {'e': [
                              ['_l'], ['_l_xy'],
                              ['q'],
                              ]}
        self._scale_feats(factor, feats_to_tfr)

        # feats rescaled by factor ** 2
        feats_to_tfr = {'v': [
                              ],
                        'e': [
                              ['_f'], ['_fxy'], 
                              ],
                        'f': [
                              ],
                        }
        self._scale_feats(factor ** 2, feats_to_tfr)

    @clone_handler
    def flip_x(self):
        """Flip points along the X axis."""
        vX = self.verts_feats(['x'])
        vX = -vX
        self.verts_feats(['x'], vX)

    @clone_handler
    def flip_y(self):
        """Flip points along the Y axis."""
        vX = self.verts_feats(['y'])
        vX = -vX
        self.verts_feats(['y'], vX)

    @clone_handler
    def recenter_xy(self):
        """Recenter points in the XY plane."""
        v_X = self.feat.v['x', 'y']
        self.feat.v['x', 'y'] = v_X - v_X.mean(dim=0, keepdim=True)

    @clone_handler
    def rescale_xyz_to_area_xy(self, area_xy, recenter=True):
        """Rescale points in the XYZ plane to a given XY area.

        Args:
            area_xy (float): The target XY area.
            recenter (bool, optional): Whether to recenter the points. Defaults to True.
        """
        if recenter:
            self.recenter_xy()
        g_a_xy = self.geoms_feat('_area_xy')
        s = area_xy / g_a_xy
        self.scale_xyz(s)

    @clone_handler
    def rescale_max_xyz_to_value(self, value=1.):
        """Rescale points in the XYZ plane to a maximum value.

        Args:
            value (float, optional): The target maximum value. Defaults to 1.
        """
        v_X_max = self.feat.v['x', 'y', 'z'].abs().max()
        s = value / v_X_max
        self.scale_xyz(s)

    @clone_handler
    def rescale_max_xy_to_value(self, value=1.):
        """Rescale points in the XY plane to a maximum value.

        Args:
            value (float, optional): The target maximum value. Defaults to 1.
        """
        v_xy_max = self.feat.v['x', 'y'].abs().max()
        s = value / v_xy_max
        self.scale_xy(s)

    @clone_handler
    def recenter_xyz(self):
        """Recenter points in the XYZ plane."""
        v_X = self.verts_feats(['x', 'y', 'z'])
        v_X = v_X - v_X.mean(dim=0, keepdim=True)
        self.verts_feats(['x', 'y', 'z'], v_X)

    @clone_handler
    def recenter_force(self):
        """Recenter force vectors."""
        if self.has_feats(type_='f', feats=['force_x', 'force_y']):
            vX_ = self.faces_feats(['force_x', 'force_y'])
            vX_ = vX_ - vX_.mean(dim=0, keepdim=True)
            self.faces_feats(['x_', 'y_'], vX_)

    @register_transform_func(name='RandomTranslateXY',
                             params=dict(pc_xy_bound=Interval(0., 0.1)))
    def random_translate_xy(self, pc_xy_bound=None):
        """Randomly translate points in the XY plane.

        Args:
            pc_xy_extents (float, optional): The percentage of XY extents. Defaults to 0.1.
            range (list, optional): The translation range. Defaults to None.
        """
        #
        v_X     = self.feat.v['x', 'y']
        v_X_rng = v_X.max(dim=0, keepdim=True)[0] - v_X.min(dim=0, keepdim=True)[0]
        v_dX    = v_X_rng * pc_xy_bound.sample(size=(1, 2))
        self.translate_xyz(v_dX.flatten().tolist() + [0.])      # FIXME tolist

    @register_transform_func(name='RandomTranslateXYZ',
                             params=dict(pc_xy_bound=Interval(0., 0.1)))
    def random_translate_xyz(self, pc_xy_bound=None):
        """Randomly translate points in the XY plane.

        Args:
            pc_xy_extents (float, optional): The percentage of XY extents. Defaults to 0.1.
            range (list, optional): The translation range. Defaults to None.
        """
        #
        v_X     = self.feat.v['x', 'y', 'z']
        v_X_rng = v_X.max(dim=0, keepdim=True)[0] - v_X.min(dim=0, keepdim=True)[0]
        v_dX    = v_X_rng * pc_xy_bound.sample(size=(1, 2))
        self.translate_xyz(v_dX.flatten().tolist() + [0.]) 

    @register_transform_func(name='RandomRotate90Deg',
                             params=dict(angle=Categories([90, 180, 270])))
    def random_rotate_90_deg(self, angle=None):
        """Randomly rotate points by 90, 180, or 270 degrees."""
        self.rotate_xy(angle.sample())

    @register_transform_func(name='RandomRotateXY',
                             params=dict(angle=Interval(-180.0, 180.0)))
    def random_rotate_xy(self, angle=None):
        """Randomly rotate points in the XY plane.

        Args:
            range (list, optional): The rotation range. Defaults to [-180, 180].

        Returns:
            CWGDesignTransformations: The transformed object.
        """
        self.rotate_xy(angle.sample().item())

    def random_scale_xyz(self, range=[0.75, 1.25]):
        """Randomly scale points in the XYZ plane.

        Args:
            range (list, optional): The scaling range. Defaults to [0.75, 1.25].

        Returns:
            CWGDesignTransformations: The transformed object.
        """
        scl = uniform(*range)
        self.scale_xyz(scl)

    def random_scale_z(self, range=[0.75, 1.25]):
        """Randomly scale points in the Z axis.

        Args:
            range (list, optional): The scaling range. Defaults to [0.75, 1.25].

        Returns:
            CWGDesignTransformations: The transformed object.

        scl = uniform(*range)

    def faces_vertices(self, fkeys=None):
        """Get the vertices of faces.

        Args:
            fkeys (list, optional): The face keys. Defaults to None.

        Returns:
            dict: The vertices of faces.
        """
        return self._halfedge['face']

    def random_vertices(self, n=None, pc=None, 
                        n_min=1, n_max=None,
                        vis_excluded=None, **conditions):
        """Get random vertices.

        Args:
            n (int, optional): The number of vertices. Defaults to None.
            pc (float, optional): The percentage of vertices. Defaults to None.
            n_min (int, optional): The minimum number of vertices. Defaults to 1.
            n_max (int, optional): The maximum number of vertices. Defaults to None.
            vis_excluded (list, optional): The excluded vertices. Defaults to None.
            **conditions: Additional conditions.

        Returns:
            list: The random vertices.
        """
        #conditions.setdefault('is_fixed', False)
        vis_excl = conditions.get('exclude', [])

        idx = self.vis()
        idx_ = set(idx.tolist())
        for _cnd_k, _cnd_v in conditions.items():
            _vals = self.verts_feat(_cnd_k)
            _idx = idx[_vals.flatten() == _cnd_v].tolist()
            idx_ = set.intersection(set(_idx), idx_)

        idx_ = idx_ - set(vis_excl)
        if vis_excluded:
            idx_ = idx_ - set(vis_excluded)

        idx = sorted(idx_)

        if pc is not None and n is None:
            n = int(pc * self.number_of_('v'))

        n = max(n_min, n)
        n = randint(1, n)

        idx = sample(idx_, k=n)
        return idx

    def ARAP_deform(self, anchors, vectors, 
                    k=3, axes_fixed=[],
                    auto_fix=True, copy=True):
        """Apply As-Rigid-As-Possible (ARAP) deformation.

        Args:
            anchors (list): The anchor points.
            vectors (list): The deformation vectors.
            k (int, optional): The number of iterations. Defaults to 3.
            axes_fixed (list, optional): The fixed axes. Defaults to [].
            auto_fix (bool, optional): Whether to automatically fix points. Defaults to True.
            copy (bool, optional): Whether to copy the object. Defaults to True.

        Returns:
            CWGDesignTransformations: The deformed object.
        """
        import igl

        v_X_0, fvis    = self.to_tri_vertices_and_faces()
        v_X_0_np, fvis_np = v_X_0.numpy(), fvis.numpy()

        vi_fxd_a        = self.vis_fixed().tolist() if auto_fix else []
        vi_fxd_u        = anchors.tolist() if isinstance(anchors, th.Tensor) else anchors
        vi_fxd_a_       = list(set(vi_fxd_a) - set(vi_fxd_u))
        vi_fxd_         = vi_fxd_u + vi_fxd_a_

        n_vi_fxd_u      = len(vi_fxd_u)
        n_vi_fxd_a_     = len(vi_fxd_a_)
        n_vi_fxd        = n_vi_fxd_u + n_vi_fxd_a_

        v_fxd_dX_np_    = np.zeros((n_vi_fxd, 3))
        v_fxd_dX_u_np_  = np.array(vectors.tolist() if isinstance(vectors, th.Tensor) else vectors)
        v_fxd_dX_np_[:n_vi_fxd_u, :] = v_fxd_dX_u_np_

        v_fxd_X_1_np_   = v_X_0_np[vi_fxd_, :] + v_fxd_dX_np_

        arap    = igl.ARAP(v_X_0_np, fvis_np, k, vi_fxd_)
        v_X_1   = th.tensor(arap.solve(v_fxd_X_1_np_, v_X_0_np))

        v_X_1   = v_X_1[:self.number_of_('v')]

        for _ax in axes_fixed:
            v_X_1[:, _ax:_ax+1] = v_X_0[:, _ax:_ax+1][:self.number_of_('v')]

        CWG_1   = self.local_var() if copy else self
        CWG_1.verts_feats(['x', 'y', 'z'], th.split(v_X_1.to(th.float32), [1] * 3, dim=1))
        CWG_1.geom_feat('_is_perturbed', th.tensor(1.))

        return CWG_1

    def random_ARAP_deform_z(self,        
                             pc_verts=0.1, verts_conditions={},
                             pc_z_range=0.25, n_max=None,
                             axes_fixed=[0, 1], k=3, copy=False, 
                             return_vis=False):
        """Apply random ARAP deformation in the Z axis.

        Args:
            pc_verts (float, optional): The percentage of vertices. Defaults to 0.1.
            verts_conditions (dict, optional): The vertex conditions. Defaults to {}.
            pc_z_range (float, optional): The percentage of Z range. Defaults to 0.25.
            n_max (int, optional): The maximum number of vertices. Defaults to None.
            axes_fixed (list, optional): The fixed axes. Defaults to [0, 1].
            k (int, optional): The number of iterations. Defaults to 3.
            copy (bool, optional): Whether to copy the object. Defaults to False.
            return_vis (bool, optional): Whether to return the vertices. Defaults to False.

        Returns:
            CWGDesignTransformations: The deformed object.
        """

        vis_hndls  = self.random_vertices(pc=pc_verts, n_max=n_max, **verts_conditions)
        n_hndls     = len(vis_hndls)

        v_z         = self.verts_feat('z').detach()
        v_z_max     = v_z.max()
        v_z_min     = v_z.min()    
        v_z_rng     = v_z_max - v_z_min

        # vi_to_mv = np.array(sample(vi_frd.tolist(), k=n_moves))
        v_dX = np.random.normal(np.zeros((n_hndls, 1)), scale=v_z_rng * pc_z_range)

        CWG_1 = self.ARAP_deform(anchors=vis_hndls, vectors=v_dX, axes_fixed=axes_fixed, copy=copy)
        if return_vis:
            return CWG_1, vis_hndls
        return CWG_1

    def random_ARAP_deform_xy(self,        
                              pc_verts_changed=0.1, n_max_changed=5, verts_conditions={}, 
                              pc_verts_fixed=0.1, n_max_fixed=5, auto_fix=False,
                              pc_xy_range=0.1, axes_fixed=[2], k=3, copy=False):
        """
        Applies a random ARAP deformation to the object in the XY plane.

    def ARAP_deform(self, anchors, vectors, 
                k=3, axes_fixed=[],
                auto_fix=True, copy=True):
    """Apply As-Rigid-As-Possible (ARAP) deformation.

    Args:
        anchors (list): The anchor points.
        vectors (list): The deformation vectors.
        k (int, optional): The number of iterations. Defaults to 3.
        axes_fixed (list, optional): The fixed axes. Defaults to [].
        auto_fix (bool, optional): Whether to automatically fix points. Defaults to True.
        copy (bool, optional): Whether to copy the object. Defaults to True.

    Returns:
        CWGDesignTransformations: The deformed object.
    """
    import igl

    v_X_0, fvis    = self.to_tri_vertices_and_faces()
    v_X_0_np, fvis_np = v_X_0.numpy(), fvis.numpy()

    vi_fxd_a        = self.vis_fixed().tolist() if auto_fix else []
    vi_fxd_u        = anchors.tolist() if isinstance(anchors, th.Tensor) else anchors
    vi_fxd_a_       = list(set(vi_fxd_a) - set(vi_fxd_u))
    vi_fxd_         = vi_fxd_u + vi_fxd_a_

    n_vi_fxd_u      = len(vi_fxd_u)
    n_vi_fxd_a_     = len(vi_fxd_a_)
    n_vi_fxd        = n_vi_fxd_u + n_vi_fxd_a_

    v_fxd_dX_np_    = np.zeros((n_vi_fxd, 3))
    v_fxd_dX_u_np_  = np.array(vectors.tolist() if isinstance(vectors, th.Tensor) else vectors)
    v_fxd_dX_np_[:n_vi_fxd_u, :] = v_fxd_dX_u_np_

    v_fxd_X_1_np_   = v_X_0_np[vi_fxd_, :] + v_fxd_dX_np_

    arap    = igl.ARAP(v_X_0_np, fvis_np, k, vi_fxd_)
    v_X_1   = th.tensor(arap.solve(v_fxd_X_1_np_, v_X_0_np))

    v_X_1   = v_X_1[:self.number_of_('v')]

    for _ax in axes_fixed:
        v_X_1[:, _ax:_ax+1] = v_X_0[:, _ax:_ax+1][:self.number_of_('v')]

    CWG_1   = self.local_var() if copy else self
    CWG_1.verts_feats(['x', 'y', 'z'], th.split(v_X_1.to(th.float32), [1] * 3, dim=1))
    CWG_1.geom_feat('_is_perturbed', th.tensor(1.))

    return CWG_1

    #conditions.setdefault('is_fixed', False)
    vis_excl = conditions.get('exclude', [])

    idx = self.vis()
    idx_ = set(idx.tolist())
    for _cnd_k, _cnd_v in conditions.items():
        _vals = self.verts_feat(_cnd_k)
        _idx = idx[_vals.flatten() == _cnd_v].tolist()
        idx_ = set.intersection(set(_idx), idx_)

    idx_ = idx_ - set(vis_excl)
    if vis_excluded:
        idx_ = idx_ - set(vis_excluded)

    idx = sorted(idx_)

    if pc is not None and n is None:
        n = int(pc * self.number_of_('v'))

    n = max(n_min, n)
    n = randint(1, n)

    idx = sample(idx_, k=n)
    return idx

    def transform_points(transformation, points):
    """Transform points using a given transformation matrix.

    Args:
        transformation (torch.Tensor): The transformation matrix.
        points (torch.Tensor): The points to be transformed.

    Returns:
        torch.Tensor: The transformed points.
    """
    is_3d = points.shape[1] == 3
    if not is_3d:
        assert points.shape[1] == 2
        points = th.cat((points, points.new_zeros((points.shape[0], 1))), dim=1)

    T, X, X_ = transformation, points, points

    if T.shape[0] > 3:
        X_ = F.pad(X, (0, 1, 0, 0), value=1.)

    X_ = th.matmul(T, X_.T).T

    if T.shape[0] > 3:
        X_ = X_[:, :3]

    if not is_3d:
        X_ = X_[:, :2]

    return X_
