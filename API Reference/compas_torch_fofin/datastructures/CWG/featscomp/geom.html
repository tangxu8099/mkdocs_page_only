    @register_feat_func(feats=['_nx', '_ny', '_nz'], 
                        elem_type='v', autoset=False, return_res=False)
    def verts_normals(self, v_X=None):
        """
        Computes the normals of the vertices.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The normals of the vertices.
        """
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']
        v_N = compute_vertices_normals(self, v_X).to(th.float32)
        self.verts_feats(['_nx', '_ny', '_nz'], values=v_N)
        return v_N

    @register_feat_func(feat='_area', elem_type='v')
    def verts_areas(self, v_X=None):
        """
        Computes the areas of the vertices.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The areas of the vertices.
        """
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']
        v_a = compute_vertices_areas(self, v_X).to(th.float32)
        self.verts_feat('_area', value=v_a)
        return v_a

    @register_feat_func(feat='_area_xy', elem_type='v')
    def verts_areas_xy(self, v_X=None):
        """
        Computes the areas of the vertices in the XY plane.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The areas of the vertices in the XY plane.
        """
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']
        v_X = th.cat((v_X[:, :2], th.zeros_like(v_X)[:, 2:]), dim=1)
        v_a = compute_vertices_areas(self, v_X).to(th.float32)
        return v_a

    @register_feat_func(feat='_area', elem_type='f')
    def faces_areas(self, v_X=None):
        """
        Computes the areas of the faces.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The areas of the faces.
        """
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']
        f_a = compute_faces_areas(self, v_X, store_feat='_area')
        return f_a

    @register_feat_func(feat='_area_xy', elem_type='f')
    def faces_areas_xy(self, v_X=None):
        """
        Computes the areas of the faces in the XY plane.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The areas of the faces in the XY plane.
        """
        v_X = self.verts_feats(['x', 'y', 'z'])
        v_X = v_X.clone()
        v_X[:, 2] *= 0.
        f_a = compute_faces_areas(self, v_X, store_feat='_area_xy')
        return f_a

    @register_feat_func(feats=['_cnx', '_cny', '_cnz'],

    def faces_centers(self, v_X=None):

    class CWGSpatialGraphGeometricFeatures:
    """
    Class for computing geometric features on a CWGraph.
    """
    @register_feat_func(feats=['_lx', '_ly', '_lz'], 
                        elem_type='v', autoset=False, return_res=False)
    def verts_laplacians(self, v_X=None):
        """
        Computes the Laplacians of the vertices.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The Laplacians of the vertices.
        """
        from compas_torch_fofin.funcs.gmp.laplacian import compute_nodes_laplacians
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']
        v_L = compute_nodes_laplacians(self, v_X, 'X', ntype='v').to(th.float32)
        self.feat.v['_lx', '_ly', '_lz'] = v_L
        return v_L 

    @register_feat_func(feats=['_ctx', '_cty', '_ctz'], 
                        elem_type='e', autoset=False, return_res=False)
    def edges_centroids(self, v_X=None):
        """
        Computes the centroids of the edges.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The centroids of the edges.
        """
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']

        self.edges_feats(['_ctx', '_cty', '_ctz'], values=e_C)
        return e_C

    @register_feat_func(feat='_l', elem_type='e', autoset=True)
    def edges_lengths(self, v_X=None, **kwargs):
        """
        Computes the lengths of the edges.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.
            **kwargs: Additional keyword arguments.

        Returns:
            torch.Tensor: The lengths of the edges.
        """
        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']
        e_l = compute_form_edges_lengths(self, v_X, **kwargs).to(th.float32)
        # if update:
        #     self.feat.e._l = e_l
        return e_l

    @register_feat_func(feat='_lxy', elem_type='e')
    def edges_lengths_xy(self, v_X=None):
        """
        Computes the lengths of the edges in the XY plane.

        Args:
            v_X (torch.Tensor, optional): The vertex coordinates. Defaults to None.

        Returns:
            torch.Tensor: The lengths of the edges in the XY plane.

        v_X = v_X if v_X is not None else self.feat.v['x', 'y', 'z']

        v_X[:, 2] *= 0.
        e_l = compute_form_edges_lengths(self, v_X).to(th.float32)
