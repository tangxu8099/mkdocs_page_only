    @unbatch_and_solve({'e_q': 'e', 'v_X': 'v', 'v_P': 'v'})
@clone_handler
def iterative_fdm(CWG, 
                  e_q=None, v_X=None, v_P=None,
                  xy_constrained=True,
                  apply_selfweight=False, max_sw_to_area_ratio=10.,
                  tol_v_X_diff=1e-5, tol_v_ta_diff=1e-4,
                  min_k=3, max_k=10, 
                  disp=False, **kwargs):
    """
    Performs an iterative form-finding process using the Force Density Method (FDM) with optional self-weight considerations.

    Args:
        CWG (object): An object representing the constrained geometry.
        e_q (torch.Tensor, optional): A tensor representing force densities for each edge.
        v_z (torch.Tensor, optional): A tensor of z-coordinate adjustments for vertices.
        v_ta_0 (torch.Tensor, optional): A tensor representing initial tributary areas for self-weight.
        v_P_live (torch.Tensor, optional): A tensor representing live external forces on vertices.
        xy_constrained (bool, optional): Whether to constrain the solution to the xy-plane. Defaults to True.
        apply_selfweight (str, optional): Specifies how self-weight should be applied ('dynamic', 'init', or None). Defaults to 'dynamic'.
        max_selfweight_fac (float, optional): Maximum factor for self-weight adjustments. Defaults to 10.
        tol_v_X_diff (float, optional): Convergence tolerance for vertex displacement differences. Defaults to 1e-5.
        tol_v_ta_diff (float, optional): Convergence tolerance for self-weight differences. Defaults to 1e-4.
        min_k (int, optional): Minimum number of iterations. Defaults to 3.
        max_k (int, optional): Maximum number of iterations. Defaults to 10.
        disp (bool, optional): If True, prints detailed progress during iterations. Defaults to False.
        update (bool, optional): If True, updates the original geometry with the results. Defaults to True.
        **kwargs: Additional keyword arguments for customization.

    Returns:
        DataObject: A result object containing updated geometry and force densities after solving.
    """

    if apply_selfweight in [None, False, 'init']:
        min_k, max_k = 1, 1

    v_X_0           = v_X if v_X is not None else CWG.feat.v[('x', 'y', 'z')]

    # process loading
    if v_P is None:
        v_P         = CWG.feat.v['px', 'py', 'pz']
    elif v_P is not None and v_P.size(1) == 1:
        v_P         = th.cat((th.zeros_like(v_X_0[:, :2]), v_P[:, 2:]), dim=-1) 

    if apply_selfweight:
        # process stablising upper limit of applied load
        g_area_xy_0     = CWG.geom_area_xy()
        g_sw_max        = g_area_xy_0 * max_sw_to_area_ratio

    if disp:
        print('...[Iterative FDM]...')

    # force density processing
    if e_q is None:
        e_q = CWG.feat.e.q
    ee_q = e_q

    if e_q.size(0) == CWG.n_ee:
        e_q = th.zeros((CWG.n_e, 1), device=CWG.device)
        e_q[CWG.feat.e._is_edge.flatten()] = ee_q
    elif e_q.size(0) == CWG.n_e:
        ee_q = e_q[CWG.feat.e._is_edge]
    else:
        raise RuntimeError('Unknown size of e_q given...')

    k = 0
    while True:
        # LOADING CONDITION

        # COMPUTE LOADS
        v_Pff = th.zeros_like(v_X_0)
        if apply_selfweight:
            v_sw                = CWG.verts_tributary_areas(v_X_0) 
            g_sw                = v_sw.sum()
            # self-weight stabliser
            g_sw_mul = 1.0 
            if max_sw_to_area_ratio is not None:
                g_sw_           = min(g_sw, g_sw_max)  
                g_sw_mul        = g_sw_ / g_sw
            v_sw_ff = th.zeros_like(v_X_0)
            v_sw_ff[:, 2:3] = - v_sw * g_sw_mul
            v_Pff += v_sw_ff

        v_Pff += v_P

        # SOLVE EQUATION
        v_X_1 = _fdm(CWG, ee_q, v_X_0, v_Pff)

        v_X_diff = (th.linalg.norm(v_X_1.detach() - 
                    v_X_0, dim=1)**2).mean()
        if disp:
            print('..k: {}, Δv_x: {:3f}'.format(k, v_X_diff.detach().item())) 

        k += 1
        if k >= min_k and \
           ((v_X_diff < tol_v_X_diff) or k >= max_k):
            break

        v_X_0 = v_X_1

    # calculate discrepency between selfweight
    v_sw_diff = th.tensor(0., device=CWG.device)
    if apply_selfweight:
        v_sw_diff = (v_sw[CWG.vis_freed] - v_sw_ff[CWG.vis_freed]).sum()
        if disp:
            print('v_sw_diff: {}'.format(v_sw_diff))

    # update information
    CWG.feat.v[('x', 'y', 'z')] = v_X_1
    CWG.feat.v[('pffx', 'pffy', 'pffz')] = v_Pff
    CWG.feat.e.q    = e_q
    CWG.feat.e._l   = CWG.edges_lengths()
    CWG.feat.e._f   = CWG.feat.e._l * e_q
